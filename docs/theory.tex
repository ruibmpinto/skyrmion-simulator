\section{Magnetic Skyrmions in Synthetic Antiferromagnets}
%======================================================================

Magnetic skyrmions are topologically non-trivial chiral spin textures
with a topological charge $S = \pm 1$. In a skyrmion, the
magnetization rotates smoothly from one out-of-plane direction at the
core to the opposite direction in the surrounding domain.

A synthetic antiferromagnet (SAF) consists of two ferromagnetic layers
(Co) separated by a non-magnetic spacer (Ru), coupled
antiferromagnetically via the Ruderman--Kittel--Kasuya--Yosida (RKKY)
interaction. In a SAF, skyrmions in the two layers have opposite core
polarities and opposite topological charges, leading to a cancellation
of the Magnus force and a vanishing skyrmion Hall
effect~\cite{pham2024}.

The SAF stack modeled here is Pt/Co/Ru/Pt/Co/Ru, following the
experimental system in~\cite{pham2024}.

%----------------------------------------------------------------------
\subsection{Notation}
\label{sec:notation}

\begin{itemize}[leftmargin=2em]
    \item $\mb{M}(\mb{r}, t)$ --- magnetization.
    \item $\Ms$ --- saturation magnetization at the operating
        temperature, held fixed.
    \item $\mb{m} = \mb{M}/\Ms$ --- unit magnetization vector,
        $|\mb{m}| = 1$; evolved by the LLGS equation.
    \item $\Heff, \Hex, \Hdmi, \Hanis, \Hdemag, \Hrkky, \Hext$ ---
        effective fields entering the LLGS equation
        $d\mb{m}/dt = -\gamma\,\mb{m}\times\Heff$. The
        $\mb{B}=\mu_0\mb{H}$ factor is absorbed into the prefactors
        (B-field units; see Sec.~\ref{sec:units}).
    \item $J_{\text{RKKY}}$ --- RKKY interlayer coupling constant;
        surface energy density (J/m$^2$) in Section~2.6, volume
        energy density (J/m$^3$) in Sec.~\ref{sec:variational},
        related by $J_{\text{RKKY}}^{(\text{vol})}
        = J_{\text{RKKY}}^{(\text{surf})}/\tCo$.
    \item $H_{\text{RKKY}}$ --- RKKY field magnitude:
        $H_{\text{RKKY}} = J_{\text{RKKY}}^{(\text{vol})}/\Ms
        = J_{\text{RKKY}}^{(\text{surf})}/(\Ms\,\tCo)$.
    \item $\mb{J} = J(t)\,\hat{J}$ --- current density;
        $J(t)$ is the magnitude, prescribed as a current pulse
        (a function of time), and $\hat{J}$ the unit vector along
        the current flow.
    \item $\phat$ --- spin polarization direction,
        $\phat = \zhat \times \hat{J}$.
    \item $\zhat$ --- film-normal unit vector (out-of-plane).
    \item $H_{\text{DL}} = \chi_{\text{DL}} J(t)$,
        $H_{\text{FL}} = \chi_{\text{FL}} J(t)$ --- damping-like and
        field-like SOT effective fields.
    \item $\Keff = K - \mu_0 \Ms^2 / 2$ --- effective uniaxial
        anisotropy with thin-film shape correction.
    \item $V_{\text{cell}} = \tCo\,a^2$ --- per-site simulation
        volume.
    \item Energy integrals over a layer:
        $\int \cdots\,dV = \tCo \int \cdots\,dS$ for thin uniform
        layers.
\end{itemize}

%======================================================================
\section{Hamiltonian}
%======================================================================

The total magnetic energy of the SAF system is the sum of
contributions from both layers
($\ell = \text{top}, \text{bot}$) plus the interlayer coupling:
%
\begin{equation}
\mathcal{H} = \sum_{\ell} \left(
    \mathcal{H}_{\text{ex}}^{(\ell)}
    + \mathcal{H}_{\text{DMI}}^{(\ell)}
    + \mathcal{H}_{\text{anis}}^{(\ell)}
    + \mathcal{H}_{\text{demag}}^{(\ell)}
    + \mathcal{H}_{\text{Z}}^{(\ell)}
\right)
+ \mathcal{H}_{\text{RKKY}}
\end{equation}

The effective field entering the LLGS equation is obtained from the
variational derivative of the energy with respect to the
magnetization:
%
\begin{equation}
\Heff = -\frac{1}{\Ms}
    \frac{\delta \mathcal{H}}{\delta \mb{m}}
\end{equation}

%----------------------------------------------------------------------
\subsection{Exchange Energy}

The Heisenberg exchange favors parallel alignment of neighboring
spins:
%
\begin{equation}
\mathcal{H}_{\text{ex}} = \Aex \int \left[
    (\nabla m_x)^2 + (\nabla m_y)^2 + (\nabla m_z)^2
\right] dV
\end{equation}

On the discrete 2D lattice with lattice constant $a$ and film
thickness $\tCo$:
%
\begin{equation}
\mathcal{H}_{\text{ex}} = -2\Aex \, \tCo
    \sum_{\langle i,j \rangle} \mb{m}_i \cdot \mb{m}_j
\end{equation}
%
where the sum runs over nearest-neighbor pairs, each counted once
(constant terms dropped).  Expanding
$|\nabla\mb{m}|^2 \approx \sum_\delta |\mb{m}_{i+\delta}
- \mb{m}_i|^2/(2a^2)$ and using $|\mb{m}|=1$ gives the factor~2.

%----------------------------------------------------------------------
\subsection{Dzyaloshinskii--Moriya Interaction Energy}

The interfacial (N\'{e}el-type) DMI at the Pt/Co interface introduces
chirality:
%
\begin{equation}
\mathcal{H}_{\text{DMI}} = D \, \tCo \int \left[
    m_z \, (\nabla \cdot \mb{m})
    - (\mb{m} \cdot \nabla) m_z
\right] dS
\end{equation}

On the discrete lattice this becomes a sum over nearest-neighbor bonds
with bond-direction-dependent cross products:
%
\begin{equation}
\mathcal{H}_{\text{DMI}} = -D \, \tCo \, a
    \sum_{\langle i,j \rangle}
    (\hat{r}_{ij} \times \zhat) \cdot
    (\mb{m}_i \times \mb{m}_j)
\end{equation}
%
where $\hat{r}_{ij}$ is the unit vector from site $i$ to site $j$
and each bond is counted once.  The factor $a$ restores energy
units: each bond carries a gradient $\sim 1/a$ over a cell area
$a^2$.

%----------------------------------------------------------------------
\subsection{Anisotropy Energy}

Uniaxial perpendicular magnetic anisotropy (PMA) along the film
normal:
%
\begin{equation}
\mathcal{H}_{\text{anis}} = -K \, \tCo \int m_z^2 \, dS
\end{equation}

On the lattice:
%
\begin{equation}
\mathcal{H}_{\text{anis}} = -K \, \tCo \, a^2 \sum_i m_{z,i}^2
\end{equation}

PMA in Co/Pt and Co/Ru/Pt stacks originates from interfacial
spin-orbit coupling at the heavy-metal interface (3d--5d
hybridization between Co and Pt). The effect scales with the
interface area, so thinner Co (here $\tCo = 1.3$~nm) has a larger
interface-to-volume ratio and stronger PMA per unit volume.

%----------------------------------------------------------------------
\subsection{Demagnetization Energy}

In the thin-film limit ($\tCo \ll$ lateral dimensions), the
magnetostatic energy reduces to a local shape anisotropy:
%
\begin{equation}
\mathcal{H}_{\text{demag}} = \frac{\mu_0 \Ms^2}{2} \, \tCo
    \int m_z^2 \, dS
\end{equation}

This opposes out-of-plane magnetization.  Since both the anisotropy
and demagnetization energies are proportional to $m_z^2$, they merge
into a single effective anisotropy:
%
\begin{equation}
\mathcal{H}_{\text{anis}} + \mathcal{H}_{\text{demag}} =
    -\underbrace{\left(K - \frac{\mu_0 \Ms^2}{2}\right)}_{\Keff}
    \, \tCo \int m_z^2 \, dS
\end{equation}
%
Three regimes arise depending on the sign of $\Keff$:
%
\begin{itemize}[nosep]
\item $K > \mu_0 \Ms^2/2$: $\Keff > 0$
    --- anisotropy wins, magnetization points out-of-plane.
\item $K < \mu_0 \Ms^2/2$: $\Keff < 0$
    --- demagnetization wins, magnetization lies in-plane.
\item $K \approx \mu_0 \Ms^2/2$: $\Keff \approx 0$
    --- spin reorientation transition (SRT); domain walls and
    skyrmions cost very little energy to form.
\end{itemize}
%
For the Co/Pt system in~\cite{pham2024},
$\mu_0 \Ms^2/2 \approx 1.285 \times 10^6$~J/m$^3$.  The measured
top-layer anisotropy $K \approx 1.294 \times 10^6$~J/m$^3$
(anisotropy field $\mu_0 H_k = 12.4$~mT) gives
$\Keff \approx 9 \times 10^3$~J/m$^3$; the value used in the
paper's simulations and in this simulator,
$K_{\text{top}} = 1.3106 \times 10^6$~J/m$^3$
($\mu_0 H_k = 36$~mT), gives
$\Keff \approx 2.6 \times 10^4$~J/m$^3$.  Either way $\Keff$ is
barely positive, right at the SRT, and this is why the system can
host skyrmions.

%----------------------------------------------------------------------
\subsubsection*{Interface PMA (Perpendicular Magnetic
Anisotropy) vs shape anisotropy: the spin reorientation
transition}

In a thin ferromagnetic film, three anisotropy sources coexist:
\begin{enumerate}[leftmargin=2em, nosep]
    \item \textbf{Volume (bulk crystal) anisotropy}
        ($K_{\text{vol}}$, J/m$^3$) --- intrinsic
        magnetocrystalline plus magneto-elastic anisotropy
        of the Co bulk, per unit volume, independent of
        layer thickness.  It is grouped together with the
        interfacial $2K_s/t$ into the single bare anisotropy
        $K = K_{\text{vol}} + 2K_s/t$ carried by the simulator.
        Sputter-deposited Co/Pt multilayers give
        $K_{\text{vol}} \approx -1.0 \times 10^6$~J/m$^3$
        \cite{lin1991}.
    \item \textbf{Interfacial PMA} ($K_s$, J/m$^2$) --- created
        by spin-orbit hybridization at heavy-metal interfaces
        (Pt, Ir, W, Pd, Au).  Energy per unit \emph{interface
        area}; spread over the layer volume each interface
        contributes $K_s/t$ per unit volume, so two equivalent
        interfaces give $2K_s/t$.  Favors out-of-plane.
    \item \textbf{Shape anisotropy}
        ($\tfrac{1}{2}\mu_0 \Ms^2$, J/m$^3$) --- from the demag
        field of a thin slab uniformly magnetized along $\zhat$.
        Per unit volume, independent of $t$.  Favors in-plane.
\end{enumerate}

The per-volume effective anisotropy of a film of thickness $t$ with
two equivalent interfaces is
%
\begin{equation}\label{eq:Keff_full}
\Keff = \underbrace{K_{\text{vol}} + \frac{2 K_s}{t}}_{K}
        - \frac{1}{2}\mu_0 \Ms^2,
\end{equation}
%
where:
\begin{itemize}[leftmargin=2em, nosep]
    \item $K_s/t$ scales as $1/t$ --- interfaces dominate as $t$
        shrinks.
    \item $\tfrac{1}{2}\mu_0 \Ms^2$ is independent of thickness ---
        pure volume term.
\end{itemize}

Sign convention:
\begin{itemize}[leftmargin=2em, nosep]
    \item $\Keff > 0 \;\Longrightarrow\;$ out-of-plane easy axis
        (PMA).
    \item $\Keff < 0 \;\Longrightarrow\;$ in-plane easy axis.
\end{itemize}

There is a critical thickness $t_c$ where $\Keff$ changes sign:
%
\begin{equation}\label{eq:t_c}
t_c = \frac{2 K_s}{\tfrac{1}{2}\mu_0 \Ms^2 - K_{\text{vol}}}.
\end{equation}
%
For Co/Pt: $K_s \sim 5\times 10^{-4}$--$10^{-3}$~J/m$^2$,
$\tfrac{1}{2}\mu_0 \Ms^2 \sim 1.3 \times 10^6$~J/m$^3$,
$K_{\text{vol}}$ tiny
$\;\Longrightarrow\; t_c \sim 1$~nm.
\begin{itemize}[leftmargin=2em, nosep]
    \item $t < t_c$: PMA wins, magnetization out-of-plane.
    \item $t > t_c$: shape anisotropy wins, magnetization in-plane.
\end{itemize}

The simulator's Co layer is $\tCo = 1.3$~nm --- \textbf{right at
the SRT, deliberately}. That is why
$\Keff = K_{\text{top}} - \mu_0 \Ms^2/2 \approx 2.6\times10^4$~J/m$^3$
is a small difference of two large numbers ($\sim 10^6$ each), and
why a 7\% uncertainty in $K$ exceeds $\Keff$ itself (see
Implementation Notes, Sec.~\ref{sec:implementation_notes}).

\begin{figure}[h!]
\centering

%----------------------------------------------------------------------
\subsection{Zeeman Energy}

Interaction with an external field $\mb{B}_{\text{ext}}$:
%
\begin{equation}
\mathcal{H}_{\text{Z}} = -\Ms \, \tCo \int
    \mb{m} \cdot \mb{B}_{\text{ext}} \, dS
\end{equation}

%----------------------------------------------------------------------
\subsection{RKKY Interlayer Coupling Energy}

The antiferromagnetic RKKY exchange between the two Co layers across
the Ru spacer:
%
\begin{equation}
\mathcal{H}_{\text{RKKY}} = J_{\text{RKKY}} \int
    \mb{m}_{\text{top}} \cdot \mb{m}_{\text{bot}} \, dS
\end{equation}
%
where $J_{\text{RKKY}} > 0$ for antiferromagnetic coupling (surface
energy density, J/m$^2$). In the Tesla convention adopted here, the
coupling field magnitude is
$H_{\text{RKKY}} = J_{\text{RKKY}} / (\Ms \, \tCo)$ (cf.\ Notation
block).

%----------------------------------------------------------------------
\subsection{Total Energy on the Discrete Lattice}

Combining all contributions, the total energy per layer on the 2D
lattice is:
%
\begin{equation}
\mathcal{H}^{(\ell)} = \tCo \, a^2 \sum_i \left[
    -\frac{\Aex}{a^2} \sum_{\delta}
        \mb{m}_i \cdot \mb{m}_{i+\delta}
    - \frac{D}{2a} \sum_{\delta}
        (\hat{\delta} \times \zhat) \cdot
        (\mb{m}_i \times \mb{m}_{i+\delta})
    - \Keff \, m_{z,i}^2
    - \Ms \, \mb{m}_i \cdot \mb{B}_{\text{ext}}
\right]
\end{equation}
%
where $\delta$ runs over the four nearest neighbors
$\{\pm\hat{x}, \pm\hat{y}\}$.  The site sum visits every bond
twice, which is why the exchange and DMI coefficients here are half
of the per-bond values above.  The interlayer RKKY term adds
$J_{\text{RKKY}} \, a^2 \sum_i
    \mb{m}_{\text{top},i} \cdot \mb{m}_{\text{bot},i}$.

%======================================================================
\section{From Hamiltonian to Effective Field}
\label{sec:variational}
%======================================================================

The Hamiltonian (Section~2) defines \emph{what} the system wants to
minimize.  The LLGS equation (Section~\ref{sec:llgs}) describes \emph{how} the
magnetization evolves in time.  The bridge between the two is the
\textbf{effective field}, obtained from the variational derivative of
the energy with respect to the magnetization.

%----------------------------------------------------------------------
\subsection{Variational Derivative}

Consider a small perturbation
$\mb{m} \to \mb{m} + \delta\mb{m}$ (subject to the constraint
$|\mb{m}| = 1$, so $\mb{m} \cdot \delta\mb{m} = 0$).  The
corresponding change in energy is:
%
\begin{equation}
\delta\mathcal{H} = \int
    \frac{\delta\mathcal{H}}{\delta\mb{m}}
    \cdot \delta\mb{m} \; dV
\end{equation}
%
The effective magnetic field is defined such that:
%
\begin{equation}\label{eq:heff_def}
\boxed{\;
\Heff = -\frac{1}{\Ms}
    \frac{\delta\mathcal{H}}{\delta\mb{m}}
\;}
\end{equation}
%
so that $\delta\mathcal{H} = -\Ms \int \Heff \cdot
\delta\mb{m} \; dV$.  The factor $1/\Ms$ gives $\Heff$ the
dimensions of magnetic field (Tesla in our convention).

%----------------------------------------------------------------------
\subsection{Derivation of the Exchange Field}

Starting from the exchange energy:
%
\begin{equation}
\mathcal{H}_{\text{ex}} = \Aex \int
    |\nabla\mb{m}|^2 \; dV
\end{equation}
%
Perturbing $\mb{m} \to \mb{m} + \delta\mb{m}$:
%
\begin{equation}
\delta\mathcal{H}_{\text{ex}} = 2\Aex \int
    \nabla\mb{m} : \nabla(\delta\mb{m}) \; dV
\end{equation}
%
where the colon denotes the tensor double contraction
$\nabla\mb{m} : \nabla(\delta\mb{m})
= \sum_\alpha \nabla m_\alpha \cdot \nabla(\delta m_\alpha)$.
Integrating by parts (boundary terms vanish for periodic boundary
conditions):
%
\begin{equation}
\delta\mathcal{H}_{\text{ex}} =
    -2\Aex \int \nabla^2\mb{m} \cdot \delta\mb{m} \; dV
\end{equation}
%
Reading off the variational derivative and applying~\eqref{eq:heff_def}:
%
\begin{equation}
\frac{\delta\mathcal{H}_{\text{ex}}}{\delta\mb{m}}
    = -2\Aex \, \nabla^2\mb{m}
\qquad\Longrightarrow\qquad
\Hex = \frac{2\Aex}{\Ms} \nabla^2\mb{m}
\end{equation}

%----------------------------------------------------------------------
\subsection{Derivation of the DMI Field}

From the interfacial DMI energy:
%
\begin{equation}
\mathcal{H}_{\text{DMI}} = D \int \left[
    m_z (\nabla \cdot \mb{m})
    - (\mb{m} \cdot \nabla) m_z
\right] dV
\end{equation}
%
Expanding each term under the variation
$\mb{m} \to \mb{m} + \delta\mb{m}$ and collecting contributions
from $\delta m_x$, $\delta m_y$, and $\delta m_z$ separately (using
integration by parts where needed):
%
\begin{equation}
\frac{\delta\mathcal{H}_{\text{DMI}}}{\delta m_x}
    = -2D\,\frac{\partial m_z}{\partial x}, \quad
\frac{\delta\mathcal{H}_{\text{DMI}}}{\delta m_y}
    = -2D\,\frac{\partial m_z}{\partial y}, \quad
\frac{\delta\mathcal{H}_{\text{DMI}}}{\delta m_z}
    = 2D\left(
        \frac{\partial m_x}{\partial x}
        + \frac{\partial m_y}{\partial y}
    \right)
\end{equation}
%
Applying~\eqref{eq:heff_def}:
%
\begin{equation}
\Hdmi = \frac{2D}{\Ms} \left(
    \frac{\partial m_z}{\partial x}\,\hat{x}
    + \frac{\partial m_z}{\partial y}\,\hat{y}
    - \left(
        \frac{\partial m_x}{\partial x}
        + \frac{\partial m_y}{\partial y}
    \right)\zhat
\right)
\end{equation}

%----------------------------------------------------------------------
\subsection{Derivation of the Anisotropy Field}

From the anisotropy energy $\mathcal{H}_{\text{anis}} = -K \int
m_z^2 \, dV$:
%
\begin{equation}
\delta\mathcal{H}_{\text{anis}} = -2K \int
    m_z \, \delta m_z \; dV
    = -2K \int (m_z \, \zhat) \cdot \delta\mb{m} \; dV
\end{equation}
%
Therefore:
%
\begin{equation}
\Hanis = \frac{2K}{\Ms} \, m_z \, \zhat
\end{equation}
%
The demagnetization field follows identically with
$K \to -\mu_0\Ms^2/2$, giving
$\Hdemag = -\mu_0\Ms \, m_z \, \zhat$, and the combined field uses
$\Keff = K - \mu_0\Ms^2/2$.

%----------------------------------------------------------------------
\subsection{Derivation of the Zeeman and RKKY Fields}

The Zeeman energy
$\mathcal{H}_{\text{Z}} = -\Ms \int \mb{m} \cdot
\mb{B}_{\text{ext}} \, dV$ is linear in $\mb{m}$:
%
\begin{equation}
\delta\mathcal{H}_{\text{Z}} = -\Ms \int
    \mb{B}_{\text{ext}} \cdot \delta\mb{m} \; dV
\qquad\Longrightarrow\qquad
\Hext = \mb{B}_{\text{ext}}
\end{equation}

Similarly, the RKKY energy
$\mathcal{H}_{\text{RKKY}} = J_{\text{RKKY}} \int
\mb{m}_{\text{top}} \cdot \mb{m}_{\text{bot}} \, dV$ gives,
when varying $\mb{m}_{\text{top}}$:
%
\begin{equation}
\left.\Hrkky\right|_{\text{top}} =
    -\frac{J_{\text{RKKY}}}{\Ms}\,\mb{m}_{\text{bot}}
    = -H_{\text{RKKY}}\,\mb{m}_{\text{bot}}
\end{equation}
%
and symmetrically for the bottom layer.

%----------------------------------------------------------------------
\subsection{Physical Interpretation of the LLG Equation}

With the effective field defined, the LLG equation
\begin{equation}
\frac{d\mb{m}}{dt} =
    \underbrace{-\gamma\,\mb{m} \times \Heff}_{\text{precession}}
    + \underbrace{\alpha\,\mb{m} \times
        \frac{d\mb{m}}{dt}}_{\text{damping}}
\end{equation}
has a clear energetic interpretation:

\begin{itemize}[nosep]
\item \textbf{Precession term}
    $(-\gamma\,\mb{m}\times\Heff)$:
    The torque is perpendicular to both $\mb{m}$ and $\Heff$,
    so the magnetization precesses around the effective field.
    Since $\mb{m} \cdot d\mb{m}/dt = 0$ (from $|\mb{m}|=1$)
    and the torque does no work on the field
    ($\Heff \cdot (\mb{m}\times\Heff) = 0$), this term
    \textbf{conserves energy}.

\item \textbf{Damping term}
    $(\alpha\,\mb{m}\times d\mb{m}/dt)$:
    This produces a torque component that tilts $\mb{m}$ toward
    $\Heff$.  One can show that the energy dissipation rate is:
    %
    \begin{equation}
    \frac{d\mathcal{H}}{dt} =
        -\frac{\alpha\Ms}{\gamma}
        \int \left|\frac{d\mb{m}}{dt}\right|^2 dV
        \leq 0
    \end{equation}
    %
    Energy is always dissipated ($d\mathcal{H}/dt \leq 0$),
    driving the system toward the energy minimum.
\end{itemize}

Together, the two terms describe a \textbf{damped precession}:
the magnetization spirals around the effective field while
gradually aligning with it.  The Hamiltonian defines the energy
landscape; the effective field is the local gradient of that
landscape; and the LLG equation is the equation of motion that
navigates the magnetization toward equilibrium.

The Slonczewski spin-transfer torque terms (Section~6) add a
non-conservative driving force from the electric current, which
can sustain steady-state motion (e.g.\ skyrmion translation)
against the damping.

\section{Effective Field Contributions}
\label{sec:effective_fields}
%======================================================================

The total effective field for each layer is:
%
\begin{equation}
\Heff = \Hex + \Hdmi + \Hanis + \Hext + \Hrkky + \Hdemag
\end{equation}
%
All fields are computed in Tesla.

\subsection{Exchange Field}

On a 2D square lattice with lattice constant $a$, the discrete
exchange field is the lattice Laplacian:
%
\begin{equation}
\Hex = \frac{2\Aex}{\Ms \, a^2}
    \left(
        \mb{m}_{+x} + \mb{m}_{-x}
        + \mb{m}_{+y} + \mb{m}_{-y}
        - 4\mb{m}
    \right)
\end{equation}

%----------------------------------------------------------------------
\subsection{Dzyaloshinskii--Moriya Interaction}

The discrete interfacial (N\'{e}el-type) DMI field:
%
\begin{align}
H_{\text{DMI},x} &= \frac{D}{\Ms \, a}
    (m_{+x,z} - m_{-x,z}) \\
H_{\text{DMI},y} &= \frac{D}{\Ms \, a}
    (m_{+y,z} - m_{-y,z}) \\
H_{\text{DMI},z} &= -\frac{D}{\Ms \, a}
    (m_{+x,x} - m_{-x,x} + m_{+y,y} - m_{-y,y})
\end{align}
%
where $D > 0$ favors left-handed N\'{e}el skyrmions, consistent with
the Pt/Co interface.

\subsection{Anisotropy and demagnetization}
\label{sec:anis_demag}

The perpendicular magnetic anisotropy (PMA) and the
demagnetization field share the same functional form
$\propto m_z\,\zhat$ in the uniform limit but originate from
completely different physics and have completely different
locality.  Throughout this subsection $\ell \in \{\text{top},
\text{bot}\}$ indexes the two Co layers of the SAF, and
$K_\ell$ is the per-layer bare anisotropy
$K_\ell = K_{\text{vol}} + 2K_s/\tCo$ of Sec.~2.3 (with
distinct values $K_{\text{top}}$ and $K_{\text{bot}}$ because
the Pt and Ru interfaces give different $K_s$).  The
codebase supports three levels of approximation, of
increasing physical fidelity:
%
\begin{itemize}[leftmargin=2em, nosep]
\item \textbf{Local $K_{\text{eff}}$ approximation}: absorbs
    the uniform-$\mb{m}$ part of the demag (i.e.\ the
    $-\tfrac{1}{2}\mu_0 \Ms^2$ shape-anisotropy energy
    density of Sec.~2.4) into a single effective constant
    $K_{\text{eff},\ell} = K_\ell - \tfrac{1}{2}\mu_0 \Ms^2$
    in the PMA term; the demag is not computed explicitly.
    Valid for textures whose in-plane variation is slow
    compared with $\tCo$ (skyrmion-scale wavelengths satisfy
    this).
\item \textbf{Analytic slab demag}: writes the full demag
    tensor analytically via the thin-film shape factor in
    Fourier space.  Uses the bare $K_\ell$ in the PMA term
    and adds the demag separately.
\item \textbf{Newell demag}: treats every lattice cell as a
    finite rectangular prism and uses the exact cell-to-cell
    demag tensor, so it captures the finite-cell corrections
    that the continuous slab smooths over.  Same bare $K_\ell$
    in the PMA term.
\end{itemize}
%
The next three subsubsections give: (i)~the discrete PMA
formula and the choice of $K_\ell$ vs $K_{\text{eff},\ell}$
in each branch; (ii)~the conceptual distinction between PMA
and demagnetization and why they look identical for uniform
$\mb{m}$; and (iii)~the explicit kernel formulas for the
three demag approaches.

%......................................................................
\subsubsection{Perpendicular magnetic anisotropy: discrete form}

Uniaxial anisotropy along the film normal ($\zhat$).  The
anisotropy field is local in space (no spatial derivative), so
its discrete form is just the continuum formula evaluated at
each lattice site $(i, j)$.  In the SAF the top and bottom Co
layers are inequivalent (Pt and Ru interfaces); the simulator
carries two distinct anisotropy constants $K_{\text{top}}$ and
$K_{\text{bot}}$:
%
\begin{align}
\mb{H}_{\text{anis},\text{top}}(i,j) &=
    \frac{2 K_{\text{top}}}{\Ms} \, m^{\text{top}}_z(i,j) \, \zhat,
    \label{eq:hanis_top} \\
\mb{H}_{\text{anis},\text{bot}}(i,j) &=
    \frac{2 K_{\text{bot}}}{\Ms} \, m^{\text{bot}}_z(i,j) \, \zhat.
    \label{eq:hanis_bot}
\end{align}
%
With the local $N_z = 1$ approximation
(Sec.~\ref{sec:demag}), the shape-anisotropy term is absorbed
into an effective constant
$K_{\text{eff},\ell} = K_\ell - \tfrac{1}{2}\mu_0 \Ms^2$
($\ell \in \{\text{top}, \text{bot}\}$), and the discrete
combined field becomes:
%
\begin{equation}
\mb{H}_{\text{anis+demag},\ell}(i,j) =
    \frac{2 K_{\text{eff},\ell}}{\Ms} \,
    m^{\ell}_z(i,j) \, \zhat.
\label{eq:hanis_keff}
\end{equation}
%
In short: in the slab and Newell branches, the two
equations~\eqref{eq:hanis_top}--\eqref{eq:hanis_bot} apply with
the bare $K_\ell$ and the demag is added separately; in the
local-$N_z = 1$ branch, equation~\eqref{eq:hanis_keff} applies
with the combined $K_{\text{eff},\ell}$ and no separate demag is
needed.  The next subsubsection explains
why this single-constant substitution is exact for
spatially uniform $\mb{m}$ and where it breaks; the kernel
formulas for the slab and Newell branches follow in
Sec.~\ref{sec:demag}.

%......................................................................
\subsubsection{PMA vs demagnetization: same form, different
physics}
\label{sec:pma_vs_demag}

PMA and the demagnetizing field happen to enter the LLGS
right-hand side as terms with the same functional form
$\propto m_z\,\zhat$, which makes them visually
indistinguishable in the uniform limit and motivates the
local $K_{\text{eff},\ell} = K_\ell -
\tfrac{1}{2}\mu_0 \Ms^2$ approximation.

\paragraph{Different physical origin.}
PMA is a \emph{quantum-mechanical interface effect}.  Strong
spin--orbit coupling at the Co/Pt and Co/Ru contacts
hybridizes Co~3d orbitals with the heavy-metal $d$~band; the
hybridization lifts the orbital degeneracy in favor of the
out-of-plane $d_{z^2}$-like orbital, and Hund coupling locks
the spin to the orbital.  The result is an interfacial easy
axis along $\zhat$, with energy density $-K_s\,m_z^2/t$ per
unit volume for a film of thickness $t$ (Sec.~2.3).

Demagnetization is a \emph{classical magnetostatic effect}.
Every magnetic moment generates a dipole field; Maxwell's
equation $\nabla \cdot \mb{B} = 0$ forces the magnetization
distribution to source surface and volume ``charges'' that
produce a self-field opposing $\mb{M}$ (Sec.~2.4).  No
spin--orbit coupling, no orbital physics --- pure classical
electromagnetism.

\paragraph{Different locality.}
The PMA field at lattice site $(i, j)$ depends only on
$m^\ell_z(i, j)$ at that same site --- it is strictly
\emph{local}, as the discrete formula above already shows.
The demagnetization field at $(i, j)$ depends on $\mb{m}$ at
\emph{every} site of \emph{both} layers through a real-space
convolution
$\Hdemag = -\mu_0 (\hat N * \mb{m})$
with a long-range demag tensor $\hat N$.  Evaluating it
requires either an FFT-based convolution (slab kernel,
Sec.~\ref{sec:demag}) or a numerical integration of magnetic
surface charges (Newell kernel, Sec.~\ref{sec:demag}).

\paragraph{Why they look identical in the uniform limit.}
For spatially uniform $\mb{m} = m_z\,\zhat$ (the same
configuration analyzed in Sec.~2.4), the demag field of an
infinite slab is purely out-of-plane and constant,
%
\begin{equation}
\Hdemag\bigl|_{\text{uniform}\ \mb{m}} =
    -\mu_0 \Ms \, m_z\,\zhat ,
\end{equation}
%
which is the field equivalent of the
$+\tfrac{1}{2}\mu_0 \Ms^2 \tCo \int m_z^2\, dS$
energy density of Sec.~2.4.  Combined with the bare PMA
field $\Hanis = (2K_\ell/\Ms)\,m_z\,\zhat$ (for layer
$\ell$):
%
\begin{equation}
\Hanis + \Hdemag\bigl|_{\text{uniform}\ \mb{m}} =
\left(\frac{2K_\ell}{\Ms} - \mu_0 \Ms\right) m_z \,\zhat
= \frac{2\,K_{\text{eff},\ell}}{\Ms}\, m_z \,\zhat,
\qquad
K_{\text{eff},\ell} = K_\ell - \tfrac{1}{2}\mu_0 \Ms^2 .
\label{eq:keff_identity}
\end{equation}
%
For uniform $\mb{m}$ the two effects are mathematically
indistinguishable from a single uniaxial anisotropy with
constant $K_{\text{eff},\ell}$.

\paragraph{Bare $K_\ell$ versus effective $K_{\text{eff},\ell}$.}
\textbf{Bare $K_\ell$} is the \emph{physically measured}
interfacial PMA constant.  It is purely interfacial in
origin and has no information about the layer's shape or
thickness baked in.  For our system:
%
\begin{equation*}
K_{\text{top}} = 1.3106\times 10^6\ \text{J/m}^3,\qquad
K_{\text{bot}} = 1.310\times 10^6\ \text{J/m}^3,
\end{equation*}
%
both $\sim 10^6$ J/m$^3$, both positive, both
\emph{out-of-plane}.

\textbf{Effective $K_{\text{eff},\ell}$} is the
\emph{algebraic combination}
$K_\ell - \tfrac{1}{2}\mu_0 \Ms^2$ that emerges from
identity~\eqref{eq:keff_identity}.  It is a numerical
artifact of the local-demag approximation, \emph{not} a
measurable quantity.  For our system,
%
\begin{equation*}
K_{\text{eff,top}} \approx 1.3106\times 10^6
- 1.285\times 10^6
\approx 2.6\times 10^4\ \text{J/m}^3 ,
\end{equation*}
%
so a 7\% uncertainty in $K_\ell$ alone exceeds $K_{\text{eff},\ell}$
(Sec.~\ref{sec:k_uncertainty}).

\paragraph{Where the local approximation breaks.}
To see where the identity \eqref{eq:keff_identity} stops
holding, decompose the in-plane magnetization texture into
Fourier modes with wavevector $\mb{k}$:
%
\begin{equation}
\mb{m}(\mb{r}) =
    \sum_{\mb{k}} \mb{m}_{\mb{k}}\, e^{i\mb{k}\cdot\mb{r}}.
\end{equation}
%
The uniform-$\mb{m}$ result of \eqref{eq:keff_identity} is
the $\mb{k} = 0$ mode of this expansion.  The slab demag
kernel component acting on $m_z$ is
%
\begin{equation}
N_{zz}^{\text{slab}}(\mb{k}) = f(|\mb{k}|\,\tCo),
\qquad
f(x) = \frac{1 - e^{-x}}{x},
\label{eq:f_shape}
\end{equation}
%
with the long- and short-wavelength limits
%
\begin{equation}
\lim_{|\mb{k}|\,\tCo \to 0} f(|\mb{k}|\,\tCo) = 1,
\qquad
\lim_{|\mb{k}|\,\tCo \to \infty} f(|\mb{k}|\,\tCo) = 0.
\end{equation}
%
The PMA field is local, so it is wavevector-independent;
combined with the demag, the per-mode out-of-plane
effective field is
%
\begin{equation}
H_z(\mb{k}) =
\left(\frac{2K_\ell}{\Ms}
      - \mu_0 \Ms\, f(|\mb{k}|\,\tCo)\right) m_z(\mb{k}).
\label{eq:Hz_per_mode}
\end{equation}
%
At $\mb{k} = 0$, $f = 1$ and \eqref{eq:Hz_per_mode} reduces
to $H_z = 2K_{\text{eff},\ell}/\Ms\,m_z(\mb 0)$ --- the
identity \eqref{eq:keff_identity}.  At short wavelengths
$|\mb{k}|\,\tCo \gg 1$, $f \to 0$ and the demag piece
vanishes:
%
\begin{equation}
\lim_{|\mb{k}|\,\tCo \to \infty} H_z(\mb{k}) =
\frac{2K_\ell}{\Ms}\, m_z(\mb{k})
\;\neq\;
\frac{2K_{\text{eff},\ell}}{\Ms}\, m_z(\mb{k}).
\end{equation}
%
Substituting $K_{\text{eff},\ell}$ for $K_\ell$ at those
wavelengths is wrong by a factor of
$2K_\ell/(2K_{\text{eff},\ell}) \sim 50$ for our parameters
($K_\ell \sim 1.3\times 10^6$ vs
$K_{\text{eff},\ell} \sim 2.6\times 10^4$~J/m$^3$).

For the present skyrmion problem the relevant length scales
are $R \sim 100$~nm and $\Delta_{\text{DW}} \sim 27$~nm, so
$|\mb{k}|\,\tCo \sim 2\pi \tCo / \Delta \sim 0.30$ at the DW
(modestly small) and $\sim 0.08$ at the skyrmion edge.  The
slab kernel is close to its long-wavelength limit there
($f \approx 1 - |\mb{k}|\tCo/2$), so the
local-$K_{\text{eff},\ell}$ approximation produces only a
few-percent error.  The slab and Newell kernels
(Sec.~\ref{sec:demag}) restore the in-plane components
$N_{xx}$, $N_{yy}$ at $|\mb{k}| > 0$, the inter-layer dipolar
coupling, and (Newell) the finite-prism corrections at
$|\mb{k}|\tCo \sim 1$ --- effects that the
local-$K_{\text{eff},\ell}$ branch discards entirely.

\paragraph{Summary of the three branches.}
\begin{itemize}[leftmargin=2em, nosep]
\item \textbf{Local-$N_z = 1$ branch}: PMA field uses
    $K_{\text{eff},\ell}$; no separate demag evaluation.
    Fast, correct in the long-wavelength uniform-$m_z$
    regime, inadequate inside the domain wall where $\mb{m}$
    is in-plane.
\item \textbf{Slab or Newell branch}: PMA field uses the
    bare $K_\ell$; demag is computed separately for every
    Fourier mode $\mb{k}$.  No double counting because the
    bare $K_\ell$ has no shape correction folded into it.
\end{itemize}

\subsubsection{Demagnetization kernels}
\label{sec:demag}

The demagnetizing (stray-dipolar) field is the long-range Coulomb-like
field generated by the magnetic ``charges'' $\rho_m = -\nabla\cdot\mb{M}$
and $\sigma_m = \mb{M}\cdot\hat{\mb{n}}$ at internal and surface
discontinuities of $\mb{M}$. Three levels of approximation are used
in the codebase, in increasing fidelity.

\begin{figure}[H]
\centering
\includegraphics[width=0.98\linewidth]{figures/demag_schematic.pdf}
\caption{Three demagnetization approaches available in the
simulator. (a) Local $\Keff$: each cell sees only the uniform-slab
self-demag $-\mu_0 \Ms m_z\,\zhat$, which is folded into an
effective anisotropy $\Keff = K - \tfrac{1}{2}\mu_0 \Ms^{2}$. No
FFT, per-site, fastest. (b) Slab kernel: each magnetic layer is
treated as a continuous slab of thickness $\tCo$ with PBC in plane;
the demag tensor in k-space follows the thin-film shape factor
$f(k, \tCo) = (1 - e^{-|k|\tCo})/(|k|\tCo)$, with self-layer and
inter-layer parts. (c) Newell kernel: each lattice cell is a finite
$a\times a \times \tCo$ rectangular prism, and the cell-to-cell
tensor is the exact field of one uniformly magnetized prism
averaged over another.}
\label{fig:demag_schematic}
\end{figure}

\paragraph{Level 0: Local $N_z = 1$ (uniform-slab limit).}
For a uniformly magnetized infinite slab of thickness $\tCo$, the
demag field is purely out-of-plane and constant:
%
\begin{equation}
\Hdemag = -\mu_0 \Ms \, m_z \, \zhat .
\end{equation}
%
Absorbed into an effective anisotropy:
%
\begin{equation}
\Keff = K - \frac{\mu_0 \Ms^2}{2}
\qquad\Longrightarrow\qquad
\mb{H}_{\text{anis+demag}} =
    \frac{2\Keff}{\Ms} \, m_z \, \zhat .
\end{equation}
%
This is the fastest option (no FFT) and is correct in the
long-wavelength uniform-$m_z$ regime, but it is \emph{wrong inside
the domain wall}, where in-plane $\mb{m}$ produces lateral
magnetic charges that the local formula ignores.

\paragraph{Level 1: Analytic slab kernel (k-space).}
Each layer is treated as a continuous thin slab of thickness $\tCo$
with periodic boundary conditions in $(x, y)$.  The demag tensor is
analytic in 2D Fourier space.  Define the slab shape factor
%
\begin{equation}
f(|k|, \tCo) = \frac{1 - e^{-|k|\,\tCo}}{|k|\,\tCo}\,,
\qquad |k| = \sqrt{k_x^2 + k_y^2},
\end{equation}
%
which interpolates between the uniform limit
$f \to 1$ at $|k|\tCo \to 0$ and the short-wave limit
$f \to 0$ at $|k|\tCo \to \infty$.  Define the structure factor
for two parallel slabs of thickness $\tCo$ separated edge-to-edge by
the non-magnetic spacer $d_{\text{Ru}}$:
%
\begin{equation}
S(|k|) =
\frac{\bigl(1 - e^{-|k|\,\tCo}\bigr)^2}{2\,|k|\,\tCo}\,
e^{-|k|\,d_{\text{Ru}}}\,.
\end{equation}
%
The self-layer tensor $\hat{N}^{\text{slab}}_{\text{self}}(\mb{k})$
acts on the source layer's own magnetization, the inter-layer tensor
$\hat{N}^{\text{slab}}_{\text{inter}}(\mb{k})$ couples it to the
other layer:
%
\begin{align}
N^{\text{self}}_{xx}(\mb{k}) &=
    \bigl(1 - f(|k|,\tCo)\bigr)\,\frac{k_x^2}{|k|^2} ,
&
N^{\text{self}}_{xy}(\mb{k}) &=
    \bigl(1 - f(|k|,\tCo)\bigr)\,\frac{k_x k_y}{|k|^2} ,
\\
N^{\text{self}}_{yy}(\mb{k}) &=
    \bigl(1 - f(|k|,\tCo)\bigr)\,\frac{k_y^2}{|k|^2} ,
&
N^{\text{self}}_{zz}(\mb{k}) &=
    f(|k|, \tCo) ,
\\
N^{\text{inter}}_{xx}(\mb{k}) &=
    S(|k|)\,\frac{k_x^2}{|k|^2} ,
&
N^{\text{inter}}_{xy}(\mb{k}) &=
    S(|k|)\,\frac{k_x k_y}{|k|^2} ,
\\
N^{\text{inter}}_{yy}(\mb{k}) &=
    S(|k|)\,\frac{k_y^2}{|k|^2} ,
&
N^{\text{inter}}_{zz}(\mb{k}) &=
    -S(|k|) .
\end{align}
%
The two layers are also coupled across components: an in-plane
texture in one layer carries volume charges $-\nabla\cdot\mb{M}$,
whose stray field has an out-of-plane part in the other layer, and
vice versa.  These cross terms have the same strength $S$ as the
other inter-layer components, shifted by a quarter period
(the factor $i$):
%
\begin{equation}
N^{\text{inter}}_{xz}(\mb{k}) = i\,\frac{k_x}{|k|}\,S(|k|),
\qquad
N^{\text{inter}}_{yz}(\mb{k}) = i\,\frac{k_y}{|k|}\,S(|k|).
\end{equation}
%
They are odd in the vertical displacement, so they change sign when
source and destination layers are swapped.  At $\mb{k} = 0$
(uniform $\mb{m}$): $N_{zz}^{\text{self}} = 1$ and all other
components vanish, recovering the local-slab limit.  Sign
convention:
%
\begin{equation}
\Hdemag(\mb{r}) = -\mu_0 \Ms\,(\hat{N}^{\text{self}} * \mb{m})(\mb{r})
                  -\mu_0 \Ms\,(\hat{N}^{\text{inter}} * \mb{m}_{\text{other}})(\mb{r}),
\end{equation}
%
where $*$ denotes 2D convolution (multiplication in $\mb{k}$).
Cost per call: 6 forward + 6 inverse FFTs.

\paragraph{Level 2: Newell numerical kernel (finite prisms).}
Each lattice cell is a uniformly magnetized rectangular prism of
size $a\times a \times \tCo$.  The cell-to-cell demag tensor is the
field that one prism's surface charges produce, averaged over the
volume of another prism.
For source magnetization $\mb{M} = M_u\,\hat{u}$ ($u \in\{x,y,z\}$),
surface charges $\pm M_u$ live on the two $u$-faces.  The
volume-averaged field at a destination cell with center-to-center
displacement $\mb{R}$ is
%
\begin{equation}
\bigl\langle H_i^{\text{Newell}}(\mb{R})\bigr\rangle_{V_{\text{dest}}}
= \frac{M_u}{V_{\text{dest}}}\!\!
  \int_{V_{\text{dest}}}\!\!\!\!d^3 r
  \left[
    \int_{+u\text{ face}}\!\!\!\!\!\frac{(\mb{r} - \mb{r}')_i}
                                          {4\pi\,|\mb{r}-\mb{r}'|^3}\,
                           dA'
   -\int_{-u\text{ face}}\!\!\!\!\!\frac{(\mb{r} - \mb{r}')_i}
                                          {4\pi\,|\mb{r}-\mb{r}'|^3}\,
                           dA'
  \right].
\label{eq:newell_integral}
\end{equation}
%
This 5-D integral (2-D source surface $\times$ 3-D destination
volume) has an exact closed form, due to Newell, Williams and
Dunlop~\cite{newell1993}: two auxiliary functions $f$ and $g$
(for diagonal and off-diagonal components) are combined in a
second-difference over the 27 corner offsets of the two prisms.
The closed form is exact but subtracts nearly equal numbers, so it
loses precision far from the source.  Beyond about 40 cells the
prism is indistinguishable from a point dipole, and the dipole
tensor is used instead.  The same integral can also be evaluated by
adaptive Gauss--Legendre quadrature (the MuMax3
recipe~\cite{mumax3}), which serves as an independent cross-check.

With periodic boundaries, every cell also feels the periodic copies
of the box.  The tensor therefore adds the contributions of the
nearest image boxes, which sit at least one box length away and are
well described by the dipole form.

For the self-cell $\mb{R} = \mb{0}$ the source and destination
coincide.  Its demag factors are given directly by Aharoni's closed
form for a rectangular prism~\cite{aharoni1998}:
%
\begin{equation}
N_c^{\text{Aharoni}}(a, b, c)
= \frac{1}{\pi}\!\Bigl[
    \tfrac{b^2 - c^2}{2bc}\!
       \log\!\tfrac{R_{abc} - a}{R_{abc} + a}
    + \tfrac{a^2 - c^2}{2ac}\!
       \log\!\tfrac{R_{abc} - b}{R_{abc} + b}
    + \tfrac{b}{2c}\!
       \log\!\tfrac{R_{ab} + a}{R_{ab} - a}
    + \cdots
  \Bigr],
\end{equation}
%
where $R_{abc} = \tfrac12\sqrt{a^2 + b^2 + c^2}$,
$R_{ab} = \tfrac12\sqrt{a^2 + b^2}$, and the remaining terms close
the formula (full expression: Aharoni~\cite{aharoni1998}, Eq.~1).
This evaluates $N^{\text{self}}_{xx, yy, zz}(\mb{R}=0)$ exactly;
off-diagonals vanish by the cell mirror symmetries.

As in the slab kernel, the inter-layer tensor includes the
$z$\,--$xy$ cross-couplings:
%
\begin{equation}
N^{\text{inter}}_{xz}, N^{\text{inter}}_{yz} \neq 0
\quad\text{(odd in the inter-layer } z\text{-displacement)} .
\end{equation}
%
These add the contributions
$-\mu_0 \Ms\bigl(N_{xz}^{\text{inter}}\, m^{\text{other}}_z\bigr)$ to
$H_x^{\text{dest}}$ (and symmetric variants); the sign flips when
swapping source and destination layers because
$N_{xz}(\mb{R}, +Z) = -N_{xz}(\mb{R}, -Z)$.

After real-space construction over all $(n_x, n_y)$ lattice
displacements, each tensor component is FFT-transformed to
$\mb{k}$-space so the per-step demag evaluation remains
6 forward $+$ 6 inverse FFTs --- the same cost as the slab
kernel.  The result is a pair of demag tensors
$\hat N^{\text{Newell,self}}(\mb{k})$ and
$\hat N^{\text{Newell,inter}}(\mb{k})$ that play the same
role as their slab counterparts but are obtained
\emph{numerically} (no closed-form
$f(|\mb{k}|\,\tCo)$ shape factor).  Schematically,
%
\begin{align}
N^{\text{Newell,self}}_{xx}(\mb{k}) &=
    \mathcal{F}\bigl\{\bar N_{xx}^{\text{self}}(\mb{R})\bigr\},
&
N^{\text{Newell,self}}_{xy}(\mb{k}) &=
    \mathcal{F}\bigl\{\bar N_{xy}^{\text{self}}(\mb{R})\bigr\}, \\
N^{\text{Newell,self}}_{yy}(\mb{k}) &=
    \mathcal{F}\bigl\{\bar N_{yy}^{\text{self}}(\mb{R})\bigr\},
&
N^{\text{Newell,self}}_{zz}(\mb{k}) &=
    \mathcal{F}\bigl\{\bar N_{zz}^{\text{self}}(\mb{R})\bigr\}, \\
N^{\text{Newell,inter}}_{xx}(\mb{k}) &=
    \mathcal{F}\bigl\{\bar N_{xx}^{\text{inter}}(\mb{R})\bigr\},
&
N^{\text{Newell,inter}}_{xy}(\mb{k}) &=
    \mathcal{F}\bigl\{\bar N_{xy}^{\text{inter}}(\mb{R})\bigr\}, \\
N^{\text{Newell,inter}}_{yy}(\mb{k}) &=
    \mathcal{F}\bigl\{\bar N_{yy}^{\text{inter}}(\mb{R})\bigr\},
&
N^{\text{Newell,inter}}_{zz}(\mb{k}) &=
    \mathcal{F}\bigl\{\bar N_{zz}^{\text{inter}}(\mb{R})\bigr\}, \\
N^{\text{Newell,inter}}_{xz}(\mb{k}) &=
    \mathcal{F}\bigl\{\bar N_{xz}^{\text{inter}}(\mb{R})\bigr\},
&
N^{\text{Newell,inter}}_{yz}(\mb{k}) &=
    \mathcal{F}\bigl\{\bar N_{yz}^{\text{inter}}(\mb{R})\bigr\},
\end{align}
%
where $\bar N_{ij}^{\text{layer}}(\mb{R})$ is the real-space
cell-cell tensor from Eq.~\eqref{eq:newell_integral} for
$\mb{R} \neq \mb{0}$ and the Aharoni override at
$\mb{R} = \mb{0}$, and $\mathcal{F}$ is the 2-D discrete
Fourier transform.  The structure is the same as the slab kernel, including the
inter-layer cross-terms $N_{xz}^{\text{inter}}$ and
$N_{yz}^{\text{inter}}$.  The sign convention is also the
same:
%
\begin{equation}
\Hdemag(\mb{r}) =
    -\mu_0 \Ms \,
    \bigl(\hat N^{\text{Newell,self}} * \mb{m}\bigr)(\mb{r})
    -\mu_0 \Ms \,
    \bigl(\hat N^{\text{Newell,inter}} * \mb{m}_{\text{other}}\bigr)(\mb{r}).
\end{equation}

\paragraph{Choosing a level.}
No level is universally best; the choice depends on the question.
At skyrmion-scale wavelengths
$|\mb{k}|\,\tCo \sim 2\pi \tCo / R \sim 0.08 \ll 1$ the slab and
Newell formulations nearly coincide, and the local level captures
the dominant shape anisotropy.  The Newell kernel matters at shorter
wavelengths, where the finite-cell corrections (in-plane components
at $|\mb{k}| > 0$ and prism-shape effects at $|\mb{k}|\,\tCo \sim
1$) become physically relevant.

\paragraph{Finite samples.}
Fourier-space multiplication is a \emph{circular} convolution: it
describes a periodic sample.  An isolated sample is modelled by
zero-padding the grid to twice its size, so the convolution no
longer wraps around.  Padding both in-plane directions describes a
finite magnet such as a disc or a rectangle; padding only across the
width describes a racetrack, periodic along its length and free at
its two edges.

\subsection{RKKY Interlayer Coupling}

The antiferromagnetic RKKY interaction promotes antiparallel
alignment between the two Co layers across the Ru/Pt spacer.
Like the PMA, it is \emph{local} in the in-plane lattice: the
field on the top spin at site $(i,j)$ depends only on the
bottom spin at the same in-plane position, and vice versa.
The discrete form is the continuum formula evaluated
site-by-site:
%
\begin{align}
\mb{H}_{\text{RKKY,top}}(i,j) &=
    -H_{\text{RKKY}} \, \mb{m}^{\text{bot}}(i,j), \\
\mb{H}_{\text{RKKY,bot}}(i,j) &=
    -H_{\text{RKKY}} \, \mb{m}^{\text{top}}(i,j),
\end{align}
%
with $H_{\text{RKKY}} > 0$ (antiferromagnetic) carried as a
single scalar in Tesla.

\begin{figure}[h!]
\centering
\includegraphics[width=0.85\linewidth]{figures/rkky_schematic.pdf}
\caption{Antiferromagnetic RKKY interlayer coupling in the SAF
stack. The Ru/Pt spacer mediates an indirect exchange
interaction between the two Co layers whose oscillating sign
is tuned to favour antiparallel alignment at the chosen
thickness $d_{\text{Ru}}+d_{\text{Pt}}=1.35$~nm. Each top spin
$\mb{m}^{\text{top}}(i,j)$ feels an effective field
$-H_{\text{RKKY}}\,\mb{m}^{\text{bot}}(i,j)$ pointing
opposite to the bottom spin at the same in-plane site, and
vice versa. The result is the AFM-locked skyrmion pair shown
on the right: identical lateral position, opposite core
polarities, opposite topological charges. The Magnus and
skyrmion-Hall contributions cancel between layers, allowing
the SAF skyrmion to move along the current direction with no
transverse drift.}
\label{fig:rkky_schematic}
\end{figure}

\section{Landau--Lifshitz--Gilbert--Slonczewski Equation (LLGS)}
\label{sec:llgs}
%======================================================================

The magnetization dynamics of each layer are governed by the LLGS
equation. For the unit magnetization vector
$\mb{m} = \mb{M}/\Ms$:
%
\begin{equation}\label{eq:llgs}
\frac{d\mb{m}}{dt} =
    -\gamma \, \mb{m} \times \Heff
    + \alpha \, \mb{m} \times \frac{d\mb{m}}{dt}
    + \gamma H_{\text{DL}}\,
        \mb{m} \times (\phat \times \mb{m})
    + \gamma H_{\text{FL}}\,
        \phat \times \mb{m}
\end{equation}
%
where:
\begin{itemize}[nosep]
\item $\gamma$ is the gyromagnetic ratio (rad/s/T),
\item $\alpha$ is the Gilbert damping constant,
\item $\Heff$ is the effective magnetic field (Tesla),
\item $H_{\text{DL}} = \chi_{\text{DL}} \cdot J(t)$ is the
    damping-like (DL) SOT field,
\item $H_{\text{FL}} = \chi_{\text{FL}} \cdot J(t)$ is the
    field-like (FL) SOT field,
\item $\phat$ is the spin polarization direction
    ($\phat = \zhat \times \hat{J}$),
\item $\mb{J} = J(t)\,\hat{J}$ is the current density: $J(t)$ is
    the magnitude (A/m$^2$), given as a current pulse in time, and
    $\hat{J}$ is the unit vector along the current flow,
\item $\zhat$ is the film-normal unit vector (out-of-plane).
\end{itemize}
%
These are the standard Slonczewski torques.  Their magnitude is
proportional to $|\phat\times\mb{m}| = \sin\vartheta$, where
$\vartheta$ is the angle between $\mb{m}$ and $\phat$, so they are
largest when $\mb{m} \perp \phat$ and vanish when $\mb{m}$ is
parallel to the spin polarization.

%----------------------------------------------------------------------
\subsection{Explicit Form}
\label{sec:llgs_explicit_derivation}

The implicit form~\eqref{eq:llgs} contains $d\mb{m}/dt$ on both sides
through the Gilbert damping term. To convert it to a fully explicit
ODE solvable by Runge--Kutta, we eliminate the right-hand-side
$d\mb{m}/dt$ algebraically. The standard procedure
follows Mallinson~\cite{mallinson1987} for the LLG part and
Khvalkovskiy~\emph{et al.}~\cite{khvalkovskiy2013} for the SOT
extension.

\paragraph{Step 1: Compact form.}
Define $\mb{s} = \phat\times\mb{m}$ and group
the right-hand side of~\eqref{eq:llgs} as
%
\begin{equation}\label{eq:F_def}
\frac{d\mb{m}}{dt} = \mb{F} + \alpha\,\mb{m}\times\frac{d\mb{m}}{dt},
\qquad
\mb{F} = -\gamma\,\mb{m}\times\Heff
        + \gamma H_{\text{DL}}\,(\mb{m}\times\mb{s})
        + \gamma H_{\text{FL}}\,\mb{s}.
\end{equation}

\paragraph{Step 2: Cross with $\mb{m}$.}
Take $\mb{m}\times$ of both sides of~\eqref{eq:F_def}:
%
\begin{equation}\label{eq:cross_m}
\mb{m}\times\frac{d\mb{m}}{dt}
    = \mb{m}\times\mb{F}
        + \alpha\,\mb{m}\times\Big(\mb{m}\times\frac{d\mb{m}}{dt}\Big).
\end{equation}
%
The triple product is simplified by the BAC--CAB identity
$\mb{a}\times(\mb{b}\times\mb{c}) = \mb{b}(\mb{a}\cdot\mb{c})
- \mb{c}(\mb{a}\cdot\mb{b})$. Using $|\mb{m}|=1$ (so
$\mb{m}\cdot\mb{m}=1$) and
$\mb{m}\cdot d\mb{m}/dt = \tfrac{1}{2}d|\mb{m}|^2/dt = 0$:
%
\begin{equation}\label{eq:bac_cab_mdot}
\mb{m}\times\Big(\mb{m}\times\frac{d\mb{m}}{dt}\Big)
    = \mb{m}\Big(\mb{m}\cdot\frac{d\mb{m}}{dt}\Big)
    - \frac{d\mb{m}}{dt}(\mb{m}\cdot\mb{m})
    = -\frac{d\mb{m}}{dt}.
\end{equation}
%
Substituting~\eqref{eq:bac_cab_mdot} into~\eqref{eq:cross_m}:
%
\begin{equation}\label{eq:cross_m_solved}
\mb{m}\times\frac{d\mb{m}}{dt}
    = \mb{m}\times\mb{F} - \alpha\,\frac{d\mb{m}}{dt}.
\end{equation}

\paragraph{Step 3: Solve for $d\mb{m}/dt$.}
Insert~\eqref{eq:cross_m_solved} back into the
$\alpha\,\mb{m}\times d\mb{m}/dt$ term of~\eqref{eq:F_def}:
%
\begin{equation}
\frac{d\mb{m}}{dt}
    = \mb{F} + \alpha\big(\mb{m}\times\mb{F} - \alpha\,\tfrac{d\mb{m}}{dt}\big),
\end{equation}
%
collect the $d\mb{m}/dt$ terms on the left, and divide by
$1+\alpha^2$:
%
\begin{equation}\label{eq:llgs_F_explicit}
\boxed{\;
\frac{d\mb{m}}{dt}
    = \frac{1}{1+\alpha^2}\Big[\mb{F} + \alpha\,\mb{m}\times\mb{F}\Big].
\;}
\end{equation}

\paragraph{Step 4: Expand $\mb{m}\times\mb{F}$.}
The three contributions to $\mb{F}$ in~\eqref{eq:F_def} give:
%
\begin{align}
\mb{m}\times\big(-\gamma\mb{m}\times\Heff\big)
    &= -\gamma\,\mb{m}\times(\mb{m}\times\Heff), \\
\mb{m}\times\big(\gamma H_{\text{DL}}\,\mb{m}\times\mb{s}\big)
    &= \gamma H_{\text{DL}}\,\mb{m}\times(\mb{m}\times\mb{s})
     = -\gamma H_{\text{DL}}\,\mb{s},
        \label{eq:DL_simplify}\\
\mb{m}\times\big(\gamma H_{\text{FL}}\,\mb{s}\big)
    &= \gamma H_{\text{FL}}\,\mb{m}\times\mb{s},
\end{align}
%
where~\eqref{eq:DL_simplify} again uses BAC--CAB with
$\mb{m}\cdot\mb{s}=0$ (since $\mb{s}\propto\phat\times\mb{m}$ is
perpendicular to $\mb{m}$):
$\mb{m}\times(\mb{m}\times\mb{s}) = \mb{m}(\mb{m}\cdot\mb{s})
- \mb{s} = -\mb{s}$.

\paragraph{Step 5: Collect coefficients.}
Assembling $\mb{F} + \alpha\mb{m}\times\mb{F}$ and dividing by
$1+\alpha^2$:
%
\begin{equation}\label{eq:llgs_explicit}
\boxed{\;
\frac{d\mb{m}}{dt}
    = \frac{\gamma}{1 + \alpha^2} \Big[
        -\mb{m}\times\Heff
        - \alpha\,\mb{m}\times(\mb{m}\times\Heff)
        + (H_{\text{DL}} + \alpha H_{\text{FL}})\,(\mb{m}\times\mb{s})
        + (H_{\text{FL}} - \alpha H_{\text{DL}})\,\mb{s}
    \Big].
\;}
\end{equation}

The coefficients $H_{\text{DL}} + \alpha H_{\text{FL}}$ and
$H_{\text{FL}} - \alpha H_{\text{DL}}$ are the standard
``mixed'' SOT coefficients of the explicit
form~\cite{khvalkovskiy2013}; the cross-coupling between DL and FL
through $\alpha$ arises entirely from
Step~3's $\alpha\,\mb{m}\times\mb{F}$ term and vanishes at $\alpha=0$.
%
The implementation in \texttt{src/skyrmion\_simulator/simulator/integrator.py}
(\texttt{llgs\_rhs}) precomputes
$\gamma' = \gamma/(1+\alpha^2)$,
$c_{\text{DL}} = H_{\text{DL}} + \alpha H_{\text{FL}}$,
$c_{\text{FL}} = H_{\text{FL}} - \alpha H_{\text{DL}}$,
and evaluates~\eqref{eq:llgs_explicit} directly without further
algebra.

%----------------------------------------------------------------------
\subsection{Thermal Fluctuations}
\label{sec:thermal}

At finite temperature the magnetization is buffeted by the
thermal bath.  Following Brown, this is modelled by a random
thermal field $\mb{h}(t)$ added to $\Heff$ in the LLGS equation,
so it enters both the precession and the damping terms.  Each
Cartesian component in each cell is Gaussian white noise,
%
\begin{equation}
\langle h_i(t) \rangle = 0,
\qquad
\langle h_i(t)\, h_j(t') \rangle =
    \sigma^2\, \delta_{ij}\, \delta(t - t'),
\qquad
\sigma^2 = \frac{2\alpha\, k_B T}{\gamma\, \Ms\, V_{\text{cell}}} .
\end{equation}
%
The strength $\sigma^2$ follows from the fluctuation--dissipation
theorem: it is the value for which the dynamics relax to the
Boltzmann distribution at temperature $T$.  Larger damping means
stronger coupling to the bath, and a smaller cell volume means
larger relative fluctuations.  Over a time step $\Delta t$ the
field is constant with standard deviation $\sigma/\sqrt{\Delta t}$,
so its time integral has variance $\sigma^2\,\Delta t$, as white
noise requires.

The noise multiplies $\mb{m}$ inside the cross products, so the
stochastic equation must be read in the Stratonovich sense, the
interpretation that preserves $|\mb{m}| = 1$.  The Heun
predictor--corrector scheme converges to the Stratonovich solution:
it averages the slopes at the start of the step and at a
forward-Euler prediction, both evaluated with the \emph{same} noise
sample.

A current also heats the film.  To lowest order the temperature
rises quadratically with the current density,
$T = T_{\text{sub}} + R_{\text{th}}\, J^2$, where $T_{\text{sub}}$
is the substrate temperature and $R_{\text{th}}$ a thermal
resistance.

\section{Spin-Orbit Torques}
%======================================================================

The spin-orbit torques (SOT) are the mechanism by which an electric
current drives skyrmion motion.  They originate from the
\textbf{spin Hall effect} (SHE) in the heavy-metal Pt layers
adjacent to the ferromagnetic Co.

\subsection{Spin Hall Effect}

When an in-plane charge current density $\mb{J}$ flows through the
Pt layer, the strong spin-orbit coupling deflects spin-up and
spin-down electrons in opposite transverse directions.  This
generates a transverse \emph{spin current} that accumulates at the
Pt/Co interface.  The spin polarization direction is perpendicular
to both the current and the film normal:
%
\begin{equation}
\phat = \zhat \times \hat{J}
\end{equation}
%
For current along $\hat{x}$, the spin polarization is
$\phat = \hat{y}$.  The efficiency of the conversion from charge
current to spin current is characterized by the spin Hall angle
$\theta_{\text{SH}}$, which is encapsulated in the experimentally
measured SOT coefficients $\chi_{\text{DL}}$ and $\chi_{\text{FL}}$.

\begin{figure}[h!]
    \centering
    \includegraphics[width=0.85\linewidth]{figures/she_schematic.pdf}
    \caption{Spin Hall effect in the Pt/Co bilayer (side view in the
    $x$--$z$ plane; right-handed axes with $\yhat$ into the page).
    An in-plane charge current $\mb{J}\parallel+\xhat$ in the heavy
    metal Pt deflects spin-$+\yhat$ electrons toward $+\zhat$ and
    spin-$-\yhat$ electrons toward $-\zhat$. The transverse spin
    current accumulates spin polarization
    $\phat = \zhat\times\hat{J} = +\yhat$ at the Pt/Co interface
    (row of $\otimes$ markers), which drives the SOT torques on the
    Co magnetization.}
\label{fig:she_schematic}
\end{figure}

\subsection{SOT Torques in the LLGS Equation}

The spin current accumulated at the interface exerts two distinct
torques on the Co magnetization, which enter as the Slonczewski
terms in the LLGS equation:
%
\begin{itemize}[nosep]
\item \textbf{Damping-like (DL) torque}:
    $\tau_{\text{DL}} = H_{\text{DL}} \,
    \mb{m} \times (\phat \times \mb{m})$.
    This acts like an additional damping (or anti-damping) that
    pushes the magnetization toward or away from $\phat$.  It is
    the dominant torque driving skyrmion motion.

\item \textbf{Field-like (FL) torque}:
    $\tau_{\text{FL}} = H_{\text{FL}} \,
    (\phat \times \mb{m})$.
    This is equivalent to an effective magnetic field along $\phat$
    and is typically smaller in magnitude.
\end{itemize}

Both torques scale with $\sin\vartheta$, the sine of the angle
between $\mb{m}$ and $\phat$.  The SOT effective field magnitudes
are proportional to the applied current density, which follows the
current pulse $J(t)$:
%
\begin{equation}
H_{\text{DL}} = \chi_{\text{DL}} \cdot J(t),
\qquad
H_{\text{FL}} = \chi_{\text{FL}} \cdot J(t)
\end{equation}
%
where $\chi_{\text{DL}}$ and $\chi_{\text{FL}}$ are the SOT
coefficients (in T\,A$^{-1}$\,m$^2$) measured experimentally.
These coefficients encapsulate the spin Hall angle, the spin
diffusion length, and the interface spin transparency into a single
effective parameter.

\subsection{SOT Terms in the Explicit LLGS}

In the explicit form of the LLGS equation~\eqref{eq:llgs_explicit},
the SOT terms appear as the last two contributions:
%
\begin{equation}
\frac{d\mb{m}}{dt} = \frac{\gamma}{1+\alpha^2} \Big[
    \underbrace{
        -\mb{m} \times \mb{H}
        - \alpha\,\mb{m} \times (\mb{m} \times \mb{H})
    }_{\text{precession + damping}}
    + \underbrace{
        (H_{\text{DL}} + \alpha H_{\text{FL}})
            (\mb{m} \times \mb{s})
        + (H_{\text{FL}} - \alpha H_{\text{DL}})\,\mb{s}
    }_{\text{SHE-driven SOT}}
\Big]
\end{equation}
%
where $\mb{s} = \phat \times \mb{m}$.
Without the SHE ($J = 0$), the SOT terms vanish and the skyrmion
relaxes to its equilibrium configuration.  With nonzero current,
the DL-SOT exerts a net force on the skyrmion domain wall, driving
translational motion.  In the SAF geometry, the opposite topological
charges of the two layers cancel the Magnus force, resulting in
motion along the current direction with no skyrmion Hall
effect~\cite{pham2024}.

%----------------------------------------------------------------------
\subsection{Topological Spin Hall Torque}
\label{sec:tsh}

The SOT discussed above is an \emph{interface} effect generated in
the heavy-metal Pt layer.  A second current-driven torque exists in
the \emph{bulk} of the Co layer, originating from the Berry phase
that conduction electrons pick up as they adiabatically follow a
non-trivial spin texture~\cite{bruno2004, yin2015}.  This
\emph{topological spin Hall} (TSH) torque is the Pham~2024
mechanism and switches on whenever the
texture carries a non-zero local skyrmion-number density.

\paragraph{Topological charge density.}
Define the local skyrmion-number density (Pontryagin density):
%
\begin{equation}
N_{xy}(\mb{r}, t) =
    \mb{m} \cdot
    \bigl(\partial_x \mb{m} \times \partial_y \mb{m}\bigr) \,.
\label{eq:topo_density}
\end{equation}
%
Units of $1/\text{m}^2$.  Integrated over the layer, $(4\pi)^{-1}
\int N_{xy}\, dA$ gives the total topological charge $Q$ (Section
\ref{sec:topcharge}).  $N_{xy}$ is concentrated in the domain-wall
ring of the skyrmion and vanishes in the uniformly magnetized
background.

\paragraph{Berry-phase spin current.}
An electron with spin polarization $P$ drifting through a region
of nonzero $N_{xy}$ acquires a Berry phase that deflects its
trajectory transversely to the applied charge current.  The
resulting topological Hall current is itself spin-polarized; it
re-deposits angular momentum onto the underlying magnetization
through a spin-transfer-torque-like coupling.  The leading-order
contribution to $d\mb{m}/dt$ is~\cite{yin2015}:
%
\begin{equation}
\boxed{
\left.\frac{d\mb{m}}{dt}\right|_{\text{TSH}} =
    -\,b_j\,\lambda^2\,
    N_{xy}(\mb{r}, t)\,
    \partial_y \mb{m}
}
\label{eq:tsh_torque}
\end{equation}
%
with the spin-drift velocity
%
\begin{equation}
b_j = \frac{\mu_B}{q_e}\,\frac{P\,J(t)}{\Ms}
\qquad [\text{m/s}],
\label{eq:bj_def}
\end{equation}
%
where $\mu_B$ is the Bohr magneton, $q_e$ the elementary
charge, $P$ the conduction-electron spin polarization, $J(t)$
the applied charge current density, and $\Ms$ the saturation
magnetization.  $\lambda^2$ is the topological coupling length
squared (m$^2$); it sets the effective spin-diffusion area
relevant to the topological Hall current and is treated as a
material parameter.

\paragraph{Dimensions.}
The product
$b_j[\text{m/s}] \cdot \lambda^2[\text{m}^2] \cdot N_{xy}
[1/\text{m}^2] \cdot \partial_y\mb{m}[1/\text{m}]$ evaluates to
$1/\text{s}$, the correct unit for $d\mb{m}/dt$.

\paragraph{Direction.}
If the charge current is taken along $\hat{x}$, the topological Hall current
flows along $\hat{y}$, so the resulting spin transfer couples
to the magnetization gradient \emph{along $\hat{y}$} ---
$\partial_y\mb{m}$.  Rotating the current direction would
permute this through $\hat{x} \leftrightarrow \hat{y}$ in
\eqref{eq:tsh_torque}.

\paragraph{Where the torque acts.}
In the uniformly magnetized background $N_{xy} = 0$ and the
TSH torque vanishes identically.  Inside the skyrmion ring,
$|N_{xy}|$ peaks at the domain wall, so the TSH torque is
localized there.  The DL-SOT in
Eq.~\eqref{eq:llgs_explicit} drives the entire skyrmion;
the TSH adds a wall-localized push along the
direction set by $\partial_y \mb{m}$.  The two torques are
\textbf{independent and additive} --- TSH does not replace SOT.

\paragraph{Sign convention and net force.}
With $J > 0$ along $+\hat{x}$, $P > 0$, and a skyrmion of
charge $Q = -1$ in the top layer (the convention used in the
codebase), $N_{xy}$ is predominantly negative in the wall.
Equation~\eqref{eq:tsh_torque} then produces a transverse
push consistent with the topological Hall sign (electrons
deflect toward $+\hat{y}$, magnetization responds in $-\hat{y}$
via spin angular-momentum conservation).  In the SAF the
opposite skyrmion charges on the two layers send the TSH push
in opposite $\hat{y}$ directions; the net translational
contribution along $\hat{x}$ survives, but the transverse
components cancel just as the Magnus force does.

\paragraph{Discrete-lattice implementation.}
On the simulation grid, $\partial_{x,y}\mb{m}$ are evaluated as
second-order central differences.  $N_{xy}$ is the elementwise scalar
$\mb{m}\cdot(\partial_x\mb{m}\times \partial_y\mb{m})$.  The
TSH contribution is added to $d\mb{m}/dt$ after the LLGS
right-hand side:
%
\begin{equation*}
d\mb{m}/dt \mathrel{+}= -b_j\,\lambda^2\,
    N_{xy}\,\partial_y\mb{m}.
\end{equation*}
%
When the mechanism is disabled ($\lambda^2 = 0$), 
the SOT-only baseline is recovered.

%======================================================================
\section{Magnus Force and the Thiele Equation}
%======================================================================

The Magnus force is not a separate term in the LLGS equation --- it
\emph{emerges} from it when skyrmion motion is analyzed as a
collective coordinate problem.  Assume the skyrmion is a
rigid texture translated by its centre-of-mass coordinate
$\mb{R}(t)$:
%
\begin{equation}
\mb{m}(\mb{r}, t) = \mb{m}_0\bigl(\mb{r} - \mb{R}(t)\bigr),
\qquad
\partial_t \mb{m} = -(\mb{v}\cdot\nabla)\mb{m},
\qquad
\mb{v} \equiv \dot{\mb{R}}.
\label{eq:rigid_ansatz}
\end{equation}
%
Substituting~\eqref{eq:rigid_ansatz} into the explicit
LLGS~\eqref{eq:llgs_explicit} gives the rigid-translation
equation of motion,
%
\begin{equation}
-(\mb{v}\cdot\nabla)\mb{m}
= -\gamma\,\mb{m}\times\mb{H}_{\text{eff}}
  + \alpha\,\mb{m}\times\bigl[-(\mb{v}\cdot\nabla)\mb{m}\bigr]
  - \gamma\,\bm{\tau}_{\text{SOT}}.
\label{eq:rigid_substituted}
\end{equation}
%
Take the cross product with $\partial_i\mb{m}$ (with
$i\in\{x,y\}$) and integrate over the film plane.  Three
standard identities collapse the result:
%
\begin{align}
\int \mb{m}\cdot
    \bigl(\partial_x\mb{m}\times\partial_y\mb{m}\bigr)\,dA
    &= 4\pi Q, \\
G_k &\equiv \tfrac{1}{2}\,\varepsilon_{kij}
    \!\int \mb{m}\cdot
        \bigl(\partial_i\mb{m}\times\partial_j\mb{m}\bigr)\,dA
    = 4\pi Q\,\delta_{kz},
    \label{eq:gyrovector}\\
D_{ij} &\equiv \int
    \partial_i\mb{m}\cdot\partial_j\mb{m}\,dA,
    \label{eq:dissipation_tensor}
\end{align}
%
defining the gyrovector $\mb{G}=4\pi Q\,\zhat$ (the factor
$\tfrac12$ compensates the two orderings $ij = xy, yx$ in the
sum) and the
dissipation tensor $[\mb{D}]$.  The driving force from the
SOT is
%
\begin{equation}
F^{\text{SOT}}_i \equiv -\gamma
    \!\int \bm{\tau}_{\text{SOT}}\cdot\partial_i\mb{m}\,dA.
\label{eq:force_sot}
\end{equation}
%
Collecting the gyrotropic, dissipative, and driving terms gives
the \textbf{Thiele equation}:
%
\begin{equation}\label{eq:thiele}
\mb{F}_{\text{SOT}} + \mb{G} \times \mb{v}
    + \alpha [\mb{D}] \mb{v} = 0.
\end{equation}
%
The three terms represent:
%
\begin{itemize}[nosep]
\item $\mb{F}_{\text{SOT}}$: the driving force from the spin-orbit
    torque, directed along the current.
\item $\mb{G} \times \mb{v}$: the \textbf{Magnus force}, directed
    perpendicular to the skyrmion velocity.  The gyrovector is
    $\mb{G} = G\,\zhat$ with $G = 4\pi Q$, where $Q$ is the
    topological charge.  This force is analogous to the Magnus
    effect on a spinning ball and is a direct consequence of the
    skyrmion's non-trivial topology.
\item $\alpha[\mb{D}]\mb{v}$: the dissipative drag from Gilbert
    damping, where $[\mb{D}]$ is the dissipation tensor.
\end{itemize}
%
The Magnus force deflects the skyrmion sideways relative to the
driving force --- this is the \textbf{skyrmion Hall effect}.  From
\eqref{eq:thiele}, the skyrmion Hall angle is:
%
\begin{equation}
\tan\theta_{\text{SkHE}} = \frac{v_y}{v_x}
    = \frac{G}{\alpha D}
\end{equation}
%
where $D$ is the diagonal component of the dissipation tensor.

In a \textbf{single ferromagnetic layer} with $Q = \pm 1$, the
gyrovector $G = \pm 4\pi \neq 0$, and the skyrmion is deflected
at a significant angle to the current (experimentally up to
$60^\circ$--$85^\circ$ at high velocities~\cite{pham2024}).

In the \textbf{SAF}, the two layers carry opposite topological
charges ($Q_{\text{top}} = -Q_{\text{bot}}$), so their gyrovectors
cancel:
%
\begin{equation}
G_{\text{net}} = G_{\text{top}} + G_{\text{bot}}
    = 4\pi Q_{\text{top}} + 4\pi Q_{\text{bot}} = 0
\end{equation}
%
The Magnus forces on the two layers point in opposite directions
and cancel exactly.  The skyrmion moves straight along the current
direction with $\theta_{\text{SkHE}} = 0$.  This is the key
advantage of the SAF geometry for skyrmion-based devices, as
confirmed experimentally
($\theta_{\text{SkHE}} = 6.4^\circ \pm 16.4^\circ$~\cite{pham2024}).

\paragraph{Closed-form speeds.}
For a compensated SAF the Magnus term drops out, and the Thiele
equation reduces to a balance between the driving force and the
dissipative drag, $\mb{F} = \alpha[\mb{D}]\,\mb{v}$.  Evaluating
both for a rigid skyrmion of radius $R$ whose wall has the 1-D
profile of Appendix~\ref{app:dw}, with width $\Delta$, gives the
damping-like SOT speed~\cite{pham2024}
%
\begin{equation}
v_{\text{SOT}} =
    \frac{\pi\,\gamma\, H_{\text{DL}}\, R}
         {2\alpha\,\bigl(R/\Delta + \Delta/R\bigr)} .
\label{eq:v_sot}
\end{equation}
%
The denominator comes from the dissipation tensor.  For a large
skyrmion ($R \gg \Delta$) the speed saturates at
$\pi\gamma H_{\text{DL}}\Delta/(2\alpha)$: both the force and the
drag grow with the wall length, so their ratio becomes independent
of $R$.  The topological spin Hall torque of
Section~\ref{sec:tsh} adds, with the same drag,
%
\begin{equation}
v_{\text{TSH}} =
    -\frac{b_j\,\lambda^2}
          {\alpha\,\bigl(R/\Delta + \Delta/R\bigr)}
    \int N_{xy}^2\, dA ,
\label{eq:v_tsh}
\end{equation}
%
which is linear in $\lambda^2$.  The integral of $N_{xy}^2$
decreases as the skyrmion grows, because the topological density
spreads over a longer wall, so $v_{\text{TSH}}$ is largest for
small skyrmions and is a correction to $v_{\text{SOT}}$.

\section{Topological Charge}
\label{sec:topcharge}
%======================================================================

The topological charge (skyrmion number) is:
%
\begin{equation}
Q = \frac{1}{4\pi} \int \mb{m} \cdot \left(
    \frac{\partial \mb{m}}{\partial x} \times
    \frac{\partial \mb{m}}{\partial y}
\right) dx \, dy
\end{equation}
%
For a single skyrmion, $Q = \pm 1$. In the SAF, the two layers carry
opposite topological charges
($Q_{\text{top}} = -Q_{\text{bot}}$), leading to a net topological
charge of zero and hence no skyrmion Hall effect.

\subsection{Discretization}
\label{sec:topcharge_discrete}

On the lattice with spacing $a$ and periodic indices, the
partial derivatives are central differences:
%
\begin{equation}
\bigl(\partial_x \mb{m}\bigr)_{i,j} =
\frac{\mb{m}_{i,\,j+1} - \mb{m}_{i,\,j-1}}{2a},
\qquad
\bigl(\partial_y \mb{m}\bigr)_{i,j} =
\frac{\mb{m}_{i+1,\,j} - \mb{m}_{i-1,\,j}}{2a}.
\label{eq:topo_central_diff}
\end{equation}
%
The Pontryagin density at each site is
%
\begin{equation}
N_{xy}\bigl|_{i,j} \;=\; \mb{m}_{i,j} \cdot
\bigl(\partial_x\mb{m}\times\partial_y\mb{m}\bigr)_{i,j},
\label{eq:topo_density_discrete}
\end{equation}
%
and the integral becomes a Riemann sum over cells of area
$a^2$:
%
\begin{equation}
\boxed{
Q \;=\; \frac{a^2}{4\pi} \sum_{i=0}^{n_y-1}\sum_{j=0}^{n_x-1}
        N_{xy}\bigl|_{i,j}
}
\label{eq:Q_discrete}
\end{equation}

\section{Skyrmion Profile (Initial Condition)}
%======================================================================

The initial N\'{e}el skyrmion profile wraps the 1-D domain wall of
Appendix~\ref{app:dw} into a circle of radius $R$:
%
\begin{equation}
\theta(r) = 2\arctan\!\left(
    \exp\!\left(-\frac{r - R}{\Delta}\right)\right),
\end{equation}
%
\begin{equation}
m_x = \sin\theta \, \cos\varphi, \quad
m_y = \sin\theta \, \sin\varphi, \quad
m_z = \cos\theta
\end{equation}
%
where $r$ is the distance from the skyrmion center, $R$ is the
radius of the $m_z = 0$ contour, $\Delta$ the wall width, and
$\varphi = \text{atan2}(y - y_0, x - x_0)$ is the azimuthal angle
(N\'{e}el helicity).

The top layer has core down ($m_z = -1$ at center).  The bottom
layer is its full reversal, $\mb{m}_{\text{bot}} = -\mb{m}_{\text{top}}$:
core up, and in-plane spins pointing inward rather than outward.
Two facts make this the right partner.  The DMI energy is unchanged
under $\mb{m} \to -\mb{m}$, so both layers keep the chirality the
DMI favours.  And the two walls are antiparallel at every point, the
ground state of the antiferromagnetic RKKY coupling.  Flipping only
$m_z$ would give the bottom wall the wrong chirality and leave it
parallel to the top wall in-plane, frustrating both couplings.

\subsection*{Stability condition}

The RK4 time step must satisfy:
%
\begin{equation}
\omega_{\text{max}} \cdot \Delta t < 2.8
\end{equation}
%

field on the lattice, reached when neighbouring spins are
antiparallel.  With $C_{\text{ex}} = 2\Aex/(\Ms a^2) \approx 5.6$~T,
$H_{\text{max}} \approx 22$~T and
$\omega_{\text{max}} \cdot \Delta t \approx 0.22$, well within the
RK4 stability region.

\section{Implementation Notes}
\label{sec:implementation_notes}
%======================================================================

Four modelling choices and material-parameter uncertainties limit
how closely any simulation of this stack can match experiment.

%----------------------------------------------------------------------
\subsection{Local $N_z = 1$ demag vs full FFT demag}

The micromagnetic demagnetization tensor $\hat N(\mb{k})$ depends on
wavevector. For a thin slab of thickness $\tCo$:
%
\begin{equation}
N_{zz}(k) = f(|k|\,\tCo),
\qquad f(x) = \frac{1 - e^{-x}}{x}.
\end{equation}
%
\begin{itemize}[leftmargin=2em, nosep]
    \item $|k|\,\tCo \to 0$ (long wavelength, uniform film):
        $f \to 1$, $N_{zz} \to 1$, $N_{xx, yy} \to 0$. In real space:
        $\Hdemag = -\mu_0 \Ms\, m_z\,\zhat$ --- a \textbf{local}
        field driven only by $m_z$ at the same point.
    \item $|k|\,\tCo \to \infty$ (short wavelength): $f \to 0$, no
        demag.
\end{itemize}

The \textbf{local $N_z = 1$ approximation} assumes $N_{zz}=1$
everywhere and absorbs the demag into the anisotropy:
%
\begin{equation}
\Keff = K - \frac{1}{2}\mu_0 \Ms^2.
\end{equation}
%
The main RK4 loop in \texttt{src/skyrmion\_simulator/simulator/main.py} uses this
approximation; it never calls \texttt{demag.py}. It is dimensionally
correct only when:
\begin{enumerate}[leftmargin=2em, nosep]
    \item the texture varies slowly compared to $\tCo$ --- true here
        ($R \sim 80$~nm vs $\tCo = 1.3$~nm, $|k|\,\tCo \sim 0.016$);
    \item the magnetization is dominantly out-of-plane.
\end{enumerate}
The second assumption \textbf{fails inside the domain wall}, where
$\mb{m}$ is in-plane, creates surface magnetic charges, and induces
non-local stray fields between in-plane spins.

The full FFT demag (in \texttt{src/skyrmion\_simulator/simulator/demag.py}) computes
$\hat N(\mb{k})$ exactly via FFT, including in-plane components and
the inter-layer dipolar kernel across the Ru spacer. MuMax3 does
this by default; the simulator's main loop does not.  The
difference matters for the skyrmion size: non-local demag lowers
the in-plane DW energy, allowing a wider wall and a larger
skyrmion.

%----------------------------------------------------------------------
\subsection{DMI uncertainty: $D \pm 39\%$}

Interfacial DMI cannot be measured directly. Standard techniques:
\begin{itemize}[leftmargin=2em, nosep]
    \item \textbf{Brillouin light scattering (BLS):} asymmetric
        spin-wave dispersion $\omega(+k)\neq\omega(-k)$ caused by DMI.
        Sensitive to interface roughness and Pt thickness.
    \item \textbf{Domain-wall creep velocity vs in-plane field:}
        $v(H_x)$ asymmetry yields $D$.
    \item \textbf{Skyrmion size vs out-of-plane field:} equilibrium
        $R$ is monotone in $D/D_c$.
\end{itemize}
Each method has 10--30\% systematic uncertainty (interface quality,
sample variation, fitting models). Cross-method agreement is often
factor-of-2. Pham~\emph{et al.}~\cite{pham2024} quote
$D = 0.62 \pm 0.24$~mJ/m$^2$ ($\approx \pm 39\%$).

The skyrmion size depends strongly on $D$: near the Bogdanov--Hubert
threshold $D_c = (4/\pi)\sqrt{\Aex\,\Keff}$, the equilibrium $R$
diverges. A $\pm 39\%$ variation in $D$ shifts $D/D_c$ enough to
change $R$ by tens of nm.

%----------------------------------------------------------------------
\subsection{Anisotropy uncertainty: $K \pm 7\%$}
\label{sec:k_uncertainty}

$K$ is measured via FMR resonance fields, anomalous-Hall hysteresis,
or MOKE saturation fields. These methods are well-standardized and
agree within 5--10\% between labs. Hence $\pm 7\%$ on $K$.

The catch is that $\Keff = K - \mu_0 \Ms^2/2$ is a
\textbf{small difference of large numbers}:
%
\begin{equation}
K = 1.294\times 10^6,\quad
\mu_0 \Ms^2/2 = 1.285\times 10^6\ \text{J/m}^3
\;\Longrightarrow\;
\Keff \approx 9\times 10^3\ \text{J/m}^3.
\end{equation}
%
A 7\% error on $K$ ($\sim 9\times 10^4$~J/m$^3$) is
\textbf{10$\times$ larger than $\Keff$ itself} (and still
$3.5\times$ larger for the $\Keff \approx 2.6\times10^4$~J/m$^3$ of
the $36$~mT anisotropy used here) --- the system sits in
the spin reorientation transition regime where $\Keff$ is barely
positive. The domain-wall width $\Delta = \sqrt{\Aex/\Keff}$
depends on $1/\sqrt{\Keff}$, so few-percent errors on $K$ propagate
to large errors on $\Delta$ and the skyrmion size.

%----------------------------------------------------------------------
\subsection{Gilbert damping uncertainty: $\alpha \pm 0.04$}

Damping in Co/Pt is enhanced by \textbf{spin pumping}: precessing
magnetization in Co injects a transverse spin current into Pt, where
strong spin-orbit scattering dissipates it. This adds an interfacial
term to the bulk Co damping. Measurements:
\begin{itemize}[leftmargin=2em, nosep]
    \item \textbf{FMR linewidth} $\Delta H$ vs frequency $f$:
        slope $= \alpha \cdot 2\pi/\gamma$.
    \item \textbf{Time-resolved MOKE} precession decay.
\end{itemize}
Both give $\alpha \approx 0.14 \pm 0.04$ for the Co/Pt stack
($\sim \pm 30\%$).

In the SAF, the Magnus term cancels between layers
(Sec.~7), so the Thiele equation
reduces to a damping-limited drift:
%
\begin{equation}
v \propto \frac{1}{\alpha}.
\end{equation}
%
A $\pm 0.04$ uncertainty in $\alpha$ around 0.14 propagates to
$\pm 30\%$ in velocity.

\subsection{Time Integration}

The explicit LLGS equation~\eqref{eq:llgs_explicit} is integrated
using the classical 4th-order Runge--Kutta (RK4) method, whose
Butcher tableau is
%
\begin{equation}
\renewcommand{\arraystretch}{1.15}
\begin{array}{c|cccc}
0           &             &             &             &   \\
\tfrac12    & \tfrac12    &             &             &   \\
\tfrac12    & 0           & \tfrac12    &             &   \\
1           & 0           & 0           & 1           &   \\
\hline
            & \tfrac16    & \tfrac13    & \tfrac13    & \tfrac16
\end{array}
\end{equation}
%
At each substep, the spin configuration is renormalized to enforce
$|\mb{m}| = 1$:
%
\begin{align}
\mb{k}_1 &= \Delta t \, f(\mb{m}) \\
\mb{k}_2 &= \Delta t \, f\!\left(
    \text{norm}\!\left(
        \mb{m} + \tfrac{\mb{k}_1}{2}
    \right)\right) \\
\mb{k}_3 &= \Delta t \, f\!\left(
    \text{norm}\!\left(
        \mb{m} + \tfrac{\mb{k}_2}{2}
    \right)\right) \\
\mb{k}_4 &= \Delta t \, f\!\left(
    \text{norm}\!\left(
        \mb{m} + \mb{k}_3
    \right)\right) \\
\mb{m}^{n+1} &= \text{norm}\!\left(
    \mb{m}^n + \frac{
        \mb{k}_1 + 2\mb{k}_2 + 2\mb{k}_3 + \mb{k}_4
    }{6}\right)
\end{align}
%
Both SAF layers are evolved simultaneously, as the RKKY coupling
makes them interdependent.  At finite temperature the stochastic
equation is integrated with the Heun scheme of
Section~\ref{sec:thermal} instead.

%----------------------------------------------------------------------
\subsection{Boundary Conditions}

Periodic boundary conditions (PBC) are applied in both $x$ and $y$
directions by default, implemented via \texttt{np.roll} on the spin
arrays.  Finite samples use free edges instead: the demag is
zero-padded across the free direction (Sec.~\ref{sec:demag}), and
the exchange and DMI stencils see the Rohart--Thiaville free-edge
condition
$\partial_n\mb{m} = (D/2\Aex)\,(\hat{\mb{n}}\times\zhat)\times\mb{m}$
at the boundary (Sec.~\ref{sec:bc-comparison}).

%----------------------------------------------------------------------
\subsection{Output Format}

Spin configurations are written in LAMMPS dump format. Since OVITO
does not natively support spin degrees of freedom, the three spin
components ($m_x$, $m_y$, $m_z$) are written as force components
(\texttt{fx}, \texttt{fy}, \texttt{fz}), which OVITO can render as
vector arrows. Positions are output in nanometers.

Type~1 corresponds to the top Co layer ($z = 0$), type~2 to the
bottom Co layer ($z = -\tCo$).

\section{Relaxation Convergence and Finite-Size Independence}
\label{sec:relax-convergence}
%======================================================================

Before any finite-temperature or current-driven measurement, the
SAF skyrmion ($D=0.85$~mJ/m$^2$,
$\mu_0H_{k,\mathrm{top}}=12.4$~mT) is relaxed to its zero-current
ground state. Two
questions decide whether a relaxed configuration is a usable
reference: (i) how many integration steps are needed for the
skyrmion \emph{size} to converge, and (ii) whether the converged size
depends on the simulation box. Both are answered by relaxing the same
initial skyrmion in four boxes and tracking the size, energy, and
residual torque as a function of step.

\textbf{Setup.} $D=0.85$~mJ/m$^2$,
$\mu_0H_{k,\mathrm{top}}=12.4$~mT, single magnetisation pulse
switched off ($J=0$), overdamped relaxation ($\alpha=1$,
$dt=5\times10^{-14}$~s)
under the periodic (circular-convolution) Newell demag evaluated with
the Gauss--Legendre quadrature tensor. Each box is relaxed for
$2.5\times10^{5}$ steps in chunks of $2000$, \emph{never}
early-stopping, so the full convergence transient is recorded. After
every chunk the relaxed top-layer state is reduced to: the
largest-connected-component (LCC) ellipse axes $D_1,D_2$ (PBC-aware
second moments), the total energy $E$, and the percentiles of the
per-site tangential torque $|\mb{m}\times(\mb{m}\times\mb{H})|$
(maximum, $99$th percentile, median, RMS).

\textbf{Why $2.5\times10^{5}$ steps and not $4\times10^{4}$.}
Figure~\ref{fig:relax-convergence}(a) shows the diameter is
\emph{non-monotonic}: it overshoots to $\approx 211.7$~nm during the
initial collapse, then drifts downward over the next
$\sim 2\times10^{5}$ steps to the converged $206.5$~nm. Stopping at
$4\times10^{4}$ steps would report $\approx 210$~nm --- roughly
$1.7\%$ too large and still visibly drifting. The longer run is
required precisely because the size, not just the torque, must
plateau.

\textbf{Box-size independence.} The relaxed diameter is $206.5$~nm in
all four boxes (Table~\ref{tab:relax-box},
Fig.~\ref{fig:relax-convergence}(c)), even though the cell area grows
by $3.8\times$ from $256^2$ to $500\times500$. This is the expected
behaviour of a \emph{compensated} synthetic antiferromagnet: the two
antiparallel layers nearly cancel the net magnetostatic moment, so the
periodic images contribute negligibly and the confinement is set by the
intrinsic balance of exchange, DMI, and anisotropy rather than by the
cell. The practical consequence is that the periodic and the isolated
(zero-padded, free-boundary) demag give the same relaxed size for this
stack, so the boundary-condition choice is a question of matching the
reference paper, not of physics.

\textbf{Circular at $J=0$.} The relaxed skyrmion has $D_1=D_2$ to
within the ellipse resolution: with no spin--orbit torque there is no
preferred in-plane direction and the ground state is axisymmetric. The
elliptical ($D_1\neq D_2$) deformation studied elsewhere in this work
appears only once a current drives the skyrmion.

\textbf{The residual-torque floor is not under-relaxation.} The
maximum tangential torque does not vanish; it plateaus at
$\approx 6\times10^{-5}$ (Fig.~\ref{fig:relax-convergence}(b)). Two
facts identify this as a benign max-norm artifact rather than an
unrelaxed skyrmion. First, $\tau_{p99}\approx\max|\mb{\tau}|$ while the
median is three to five orders of magnitude smaller
(Table~\ref{tab:relax-box}): the residual is a broad, low-amplitude
field spread across the ferromagnetic background, not a torque spike
localised at the domain wall. Second, the floor does \emph{not} grow
with lattice size --- if anything it is slightly lower on the larger
boxes --- whereas a genuinely under-relaxed wall would tighten and the
peak torque would track the wall, not the background. The skyrmion
itself (where the size and energy are determined) is fully relaxed; the
floor is the maximum over an ever-larger set of near-zero background
sites.

\textbf{Demag-method robustness.} These results use the Gauss--Legendre
quadrature evaluation of the Newell tensor. Re-relaxing with the
closed-form Newell--Williams--Dunlop tensor (exact to machine
precision) yields the identical $206.5$~nm diameter, confirming that
the relaxed size is insensitive to the demag-evaluation method and that
the convergence behaviour above is a property of the physics, not of
the quadrature.

mid-expansion snapshots, not equilibria: the diameter is still rising

but because they follow the same free-expansion trajectory --- at

grown enough to feel its (different) box edge. The $256^2$ box is the
only one small enough that the skyrmion already touches the edge by
$\sim 1.2\times10^{5}$ steps, which is why it bends below the others and
its expansion rate halves.

\textbf{The residual torque is an edge/expansion transient, not bulk
under-relaxation.} $\max|\mb{\tau}|$ holds at $\sim 1.5\times10^{-3}$
(the moving wall and the open edges), but the \emph{median} torque
falls by more than an order of magnitude as the box grows
($1.6\times10^{-4}$ at $256^2$ to $8.0\times10^{-6}$ at $500^2$): the
interior is increasingly well-relaxed while the residual concentrates
at the expanding wall and the free boundary.

\textbf{Why this happens.} At $D=0.85$~mJ/m$^2$ with
$\mu_0H_{k,\mathrm{top}}=12.4$~mT the stack sits at
$D/D_c = 1.75$
($\sigma_{\mathrm{DW}} = 4\sqrt{\Aex\,\Keff} - \pi D = -1.14\times
10^{-3}$~J/m$^2 < 0$, Sec.~\ref{sec:relax-convergence}). With negative
wall energy the isolated skyrmion has no finite equilibrium radius --- it
lowers its energy by enlarging toward a stripe domain. Periodic
boundaries on all sides supply a confining image interaction that pins
it at the metastable, box-independent $206.5$~nm; removing the
confinement across the width (free top/bottom) eliminates that
restoring term and the skyrmion expands until arrested by the physical
edge. Consistently, the energy grows more negative as the skyrmion (and
its surrounding domain) enlarges.

\textbf{Consequence.} For this stack a free-edge --- or a true
finite-width track --- skyrmion size is set by confinement, not by
intrinsic energetics, and there is no $T=0$ free-relaxation equilibrium
to use as a seed. An intrinsically stable, finite skyrmion would require
either $D < D_c \approx 0.49$~mJ/m$^2$ (the lower end of the paper's
reported $D = 0.62\pm0.24$~mJ/m$^2$ band, where
$\sigma_{\mathrm{DW}}>0$) or hard-edge geometric confinement at the
intended width.

\subsection{Boundary conditions: periodic vs racetrack vs masked band}
\label{sec:bc-comparison}

The periodic bulk, the production racetrack, and the masked-band
diagnostic differ only in the top/bottom ($y$) boundary, and that
boundary is applied \emph{identically to both} long-range couplings ---
magnetostatics and exchange/DMI. Fig.~\ref{fig:bc-schematic} contrasts
the three. The $x$ direction (track length) is periodic throughout.

\centering

\textbf{Periodic} (\texttt{newell}). The demag is a circular convolution
on the $n_y\times n_x$ grid and the exchange/DMI stencil wraps $y\pm1$
across the box: the slab feels an infinite array of $y$ images, a
$2$-torus on both couplings.

\textbf{Racetrack} (\texttt{racetrack}). The full box is the track. The
demag kernel is built on the doubled $2n_y\times n_x$ zero-padded grid,
a linear convolution isolating a single finite-width stripe (no $y$
images); and the exchange/DMI stencil is cut at the box edge (rows $0$
and $n_y-1$), where ghost cells carry the Rohart--Thiaville free-edge
condition
$\partial_n\mb{m} = (D/2\Aex)\,(\hat{\mb{n}}\times\hat{\mb{z}})\times
\mb{m}$. Periodic $x$, free $y$, no mask: the track width is the
transverse box extent $L_y=n_y a$.

\textbf{Masked band} (\texttt{masked\_band}). The same racetrack demag,
but the magnetic region is a centred band of width $W<L_y$; vacuum
margins above and below carry the free edge. This imposes a track width
narrower than the box and is exercised by the diagnostic
\texttt{validate\_relax\_torque masked\_band}.

\textbf{Both couplings must be free.} Isolating only the demag (free
demag, periodic exchange) leaves the short-range torques on a torus and,
for $D>D_c$, the texture still expands without bound
(Sec.~\ref{sec:relax-freebc-y}). The complete racetrack frees both. The
long-range demag dominates the field change (here $\approx 17\%$ versus
periodic), while the exchange/DMI edge is local --- it matters once the
texture reaches the boundary (e.g.\ under the SOT Hall drift), not for a
centred skyrmion. The C++ and Python implementations are identical,
cross-validated to FFT round-off ($\sim10^{-12}$).

\section{Derivation of the Domain Wall Profile}
\label{app:dw}
%======================================================================

The skyrmion initial condition uses the one-dimensional domain wall
profile obtained from the Euler--Lagrange equation.  This appendix
derives that profile from first principles.

\subsection*{From 3D energy to 1D energy density}

The starting point is the three-dimensional micromagnetic energy for
a thin film with exchange and effective anisotropy (Sections~2.1
and~2.3--2.4):
%
\begin{equation}
E = \int_V \left[
    \Aex \left|\nabla \mb{m}\right|^2
    - \Keff \, m_z^2
\right] dV
\end{equation}
%
For a domain wall that varies only along~$x$ in a film of
thickness~$\tCo$ and cross-sectional area~$L_y \, \tCo$, the volume
integral reduces to:
%
\begin{equation}
E = L_y \, \tCo \int_{-\infty}^{+\infty} \left[
    \Aex \left|\frac{d\mb{m}}{dx}\right|^2
    - \Keff \, m_z^2
\right] dx
\end{equation}
%
We parametrize the unit magnetization by the polar angle
$\theta(x)$ and azimuthal angle $\phi(x)$:
%
\begin{equation}
\mb{m} = \big(
    \sin\theta\cos\phi, \;
    \sin\theta\sin\phi, \;
    \cos\theta
\big)
\end{equation}
%
Computing the gradient magnitude:
%
\begin{equation}
\left|\frac{d\mb{m}}{dx}\right|^2
    = \left(\frac{d\theta}{dx}\right)^{\!2}
    + \sin^2\!\theta
      \left(\frac{d\phi}{dx}\right)^{\!2}
\end{equation}
%
For a N\'{e}el wall (or Bloch wall), the azimuthal angle is
constant ($d\phi/dx = 0$), so the second term vanishes.
Substituting $m_z = \cos\theta$ and noting that
$-\Keff \cos^2\!\theta = -\Keff + \Keff \sin^2\!\theta$
(dropping the constant), the energy per unit cross-section becomes:
%
\begin{equation}
\frac{E}{L_y \, \tCo}
    = \int_{-\infty}^{+\infty} \varepsilon(x) \, dx
\end{equation}
%
with the one-dimensional energy density:
%
\begin{equation}\label{eq:dw_energy}
\boxed{\;
\varepsilon = \Aex
    \left(\frac{d\theta}{dx}\right)^{\!2}
    + \Keff \sin^2\!\theta
\;}
\end{equation}
%
The first term is the exchange cost of spatial variations in
$\theta$; the second is the anisotropy cost for deviating from the
easy-axis directions $\theta = 0$ ($m_z = +1$) or $\theta = \pi$
($m_z = -1$).

\subsection*{Euler--Lagrange equation}

Minimizing the total energy
$E = \int_{-\infty}^{+\infty} \varepsilon \, dx$
with respect to $\theta(x)$ gives the Euler--Lagrange equation:
%
\begin{equation}\label{eq:dw_EL}
2\Aex \frac{d^2\theta}{dx^2} = \Keff \sin(2\theta)
\end{equation}

\subsection*{First integral}

Multiplying both sides by $d\theta/dx$ and integrating once yields a
first integral (conservation of ``energy'' in the mechanical analogy):
%
\begin{equation}
\Aex \left(\frac{d\theta}{dx}\right)^{\!2}
    = \Keff \sin^2\!\theta
\end{equation}
%
which gives:
%
\begin{equation}\label{eq:dw_ode}
\frac{d\theta}{dx} = -\frac{\sin\theta}{\Delta},
\qquad
\Delta \equiv \sqrt{\frac{\Aex}{\Keff}}
\end{equation}
%
The negative sign ensures $\theta$ decreases from $\pi$ to $0$ as
$x$ increases (wall centered at $x = 0$).

\subsection*{Exact solution}

Separating variables in~\eqref{eq:dw_ode}:
%
\begin{equation}
\int \frac{d\theta}{\sin\theta} = -\frac{x}{\Delta} + C
\end{equation}
%
Using the identity
$\int \csc\theta \, d\theta
    = \ln|\tan(\theta/2)|$:
%
\begin{equation}
\ln\!\left|\tan\frac{\theta}{2}\right|
    = -\frac{x}{\Delta}
\end{equation}
%
where the integration constant is fixed by requiring $\theta(0)
= \pi/2$ (wall center). Solving for~$\theta$:
%
\begin{equation}\label{eq:dw_solution}
\boxed{\;
\theta(x) = 2\arctan\!\left(
    \exp\!\left(-\frac{x}{\Delta}\right)
\right)
\;}
\end{equation}

\subsection*{Properties}

The solution~\eqref{eq:dw_solution} has the following limits:
%
\begin{itemize}[nosep]
\item $x \to -\infty$: $\theta \to \pi$
    \quad ($m_z = -1$, down domain)
\item $x = 0$: $\theta = \pi/2$
    \quad ($m_z = 0$, wall center)
\item $x \to +\infty$: $\theta \to 0$
    \quad ($m_z = +1$, up domain)
\end{itemize}
%
The magnetization component $m_z = \cos\theta$ transitions
smoothly as:
%
\begin{equation}
m_z(x) = -\tanh\!\left(\frac{x}{\Delta}\right)
\end{equation}
%
confirming that $\Delta$ is the characteristic wall width.  Using
the parameters from~\cite{pham2024},
$\Delta = \sqrt{\Aex / \Keff} \approx 27$~nm.

\subsection*{Extension to a skyrmion}

For a radially symmetric skyrmion centered at the origin, the full
2D variational problem has no closed-form solution because the DMI
and the curvature of the domain wall add additional terms.  However,
the radial profile $\theta(r)$ is well approximated by wrapping the
1D wall into a circle at the skyrmion radius~$R_{sk}$:
%
\begin{equation}\label{eq:sk_profile}
\theta(r) = 2\arctan\!\left(
    \exp\!\left(-\frac{r - R_{sk}}{\Delta}\right)
\right)
\end{equation}
%
This gives:
%
\begin{itemize}[nosep]
\item $r \ll R_{sk}$: $\theta \to \pi$
    \quad (core, $m_z = -1$)
\item $r = R_{sk}$: $\theta = \pi/2$
    \quad ($m_z = 0$ contour)
\item $r \gg R_{sk}$: $\theta \to 0$
    \quad (background, $m_z = +1$)
\end{itemize}

The skyrmion radius $R_{sk}$ is not a free parameter --- it is
determined by the balance of exchange, DMI, and anisotropy energies.
In MuMax3, this equilibrium is found numerically: an arbitrary
initial guess is evolved under the LLG equation with high damping
until the system relaxes to the energy minimum.  Our simulator
performs the same relaxation during the $J = 0$ phase.  The
profile~\eqref{eq:sk_profile} simply provides a physically
motivated initial condition that is already close to equilibrium,
minimizing the relaxation time needed.

\section{The Constraint $\mb{m}\cdot d\mb{m}/dt = 0$}
\label{app:m_perp_mdot}
%======================================================================

The identity $\mb{m}\cdot d\mb{m}/dt = 0$ is used throughout
Section~\ref{sec:llgs_explicit_derivation} (and is the geometric
reason the LLGS equation preserves $|\mb{m}|=1$). It follows from two
steps.

\subsection*{Step 1: algebraic identity (always true)}

By the product rule applied to $|\mb{m}|^2 = \mb{m}\cdot\mb{m}$:
%
\begin{equation}
\frac{d|\mb{m}|^2}{dt}
    = \frac{d(\mb{m}\cdot\mb{m})}{dt}
    = \frac{d\mb{m}}{dt}\cdot\mb{m}
        + \mb{m}\cdot\frac{d\mb{m}}{dt}
    = 2\,\mb{m}\cdot\frac{d\mb{m}}{dt}.
\end{equation}
%
Hence
%
\begin{equation}\label{eq:m_dot_mdot_general}
\mb{m}\cdot\frac{d\mb{m}}{dt}
    = \frac{1}{2}\frac{d|\mb{m}|^2}{dt},
\end{equation}
%
which holds for \emph{any} vector $\mb{m}(t)$, with or without the
unit-norm constraint.

\subsection*{Step 2: apply the constraint $|\mb{m}| = 1$}

If $|\mb{m}| = 1$ at all times then $|\mb{m}|^2 = 1$ is a constant,
so
%
\begin{equation}
\frac{d|\mb{m}|^2}{dt} = \frac{d(1)}{dt} = 0.
\end{equation}
%
Substituting into~\eqref{eq:m_dot_mdot_general}:
%
\begin{equation}\label{eq:m_perp_mdot}
\boxed{\;
\mb{m}\cdot\frac{d\mb{m}}{dt} = 0.
\;}
\end{equation}

\subsection*{Geometric meaning}

The state vector $\mb{m}$ lives on the unit two-sphere
$S^2 \subset \mathbb{R}^3$. The tangent space at any point
$\mb{m} \in S^2$ is the plane perpendicular to the radial direction
$\mb{m}$ itself. So any velocity $d\mb{m}/dt$ that keeps $\mb{m}$ on
$S^2$ must be perpendicular to $\mb{m}$, which is precisely
statement~\eqref{eq:m_perp_mdot}.

\begin{center}
\begin{tabular}{ccc}
\textbf{Sphere $S^2$} & $\;$ & \textbf{Tangent plane at $\mb{m}$} \\
                     &      &                                 \\
$|\mb{m}| = 1$       &      & $d\mb{m}/dt \perp \mb{m}$       \\
\end{tabular}
\end{center}

\subsection*{Why the LLGS RHS preserves the constraint}

Every term in the LLGS equation~\eqref{eq:llgs_explicit} has the form
$\mb{m}\times(\,\cdots\,)$ --- precession, damping, and both SOT
torques are cross products with $\mb{m}$. A cross product
$\mb{m}\times\mb{X}$ is orthogonal to both factors, so for any vector
$\mb{X}$:
%
\begin{equation}
\mb{m}\cdot(\mb{m}\times\mb{X}) = 0.
\end{equation}
%
Therefore dotting the entire LLGS RHS with $\mb{m}$ gives zero,
$\mb{m}\cdot d\mb{m}/dt = 0$ is enforced by the equation of motion,
and the unit-norm constraint $|\mb{m}|=1$ is conserved
\emph{exactly} by the continuous-time dynamics. The RK4 integrator
introduces a small $\mathcal{O}(\Delta t^5)$ drift that is removed
by an explicit renormalization step
$\mb{m} \leftarrow \mb{m}/|\mb{m}|$ at the end of each step (see
\texttt{normalize()} in \texttt{src/skyrmion\_simulator/simulator/integrator.py}).
